   339→    mergeContacts: (primaryId, dupId) => {
   340→      const { contacts, entries, lists, notes, reminders, banks, recents, attachments } = get();

   380→      const nextReminders = remapOwner(reminders, (r) => persistReminder(r));

   405→    findDuplicates: () => {

   411→        for (let j = i + 1; j < live.length; j++) {

   418→            displayName(a).toLowerCase() === displayName(b).toLowerCase() && displayName(a) !== "Unknown";

   420→          const sharedEmail = b.emails.some((e) => aEmails.has(e.address.toLowerCase()));

   423→              primary: a,
   424→              dup: b,
   425→              reason: sharedPhone ? "Same phone number" : "Same name and email",
   426→            });

   428→          }
   429→        }
   430→      }

   503→    settleUp: (ownerId) => {
   504→      const balance = calc.ownerBalance(get().entries, ownerId);
   505→      if (Math.abs(balance) < 0.005) return 0;
   506→      let left = Math.abs(balance);
   507→      const ordered = get()
   508→        .entries.filter((e) => e.ownerId === ownerId && !e.deletedAt)
   509→        .sort((a, b) => a.date.localeCompare(b.date) || a.createdAt - b.createdAt);
   510→      for (const e of ordered) {
   511→        if (left < 0.005) break;
   512→        const rem = calc.entryRemaining(e);
   513→        // paying down: reduce |remaining| in the direction that zeroes the balance
   514→        const effective = e.type === "credit" ? rem : -rem;
   515→        if (Math.sign(effective) === Math.sign(balance)) continue; // same side, skip
   516→        const pay = Math.min(Math.abs(rem), left);
   517→        if (pay > 0.005) {
   518→          get().addPayment(e.id, { amount: r2Fallback(pay), date: todayStr(), auto: "settle", note: "Settle up" });
   519→          left = r2Fallback(left - pay);
   520→        }
   521→      }
   522→      return Math.abs(balance);
   523→    },

   525→    moveEntity: (kind, id, ownerId) => {

   528→        case "entry":
   529→          updateEntry(id, (e) => ({ ...e, ownerId }));
   530→          break;
   531→        case "list":
   532→          updateList(id, (l) => ({ ...l, ownerId }));
   533→          break;
   534→        case "note":
   535→          set((s) => {
   536→            const n = s.notes.find((x) => x.id === id);
   537→            if (n) {
   538→              const next = touch({ ...n, ownerId });
   539→              persistNote(next);
   540→              return { notes: s.notes.map((x) => (x.id === id ? next : x)) };
   541→            }
   542→            return {};
   543→          });
   544→          break;
   545→        case "reminder":
   546→          updateReminder(id, (r) => ({ ...r, ownerId }));
   547→          break;
   548→        case "bank":
   549→          set((s) => {
   550→            const b = s.banks.find((x) => x.id === id);
   551→            if (b) {
   552→              const next = touch({ ...b, ownerId });
   553→              persistBank(next);
   554→              return { banks: s.banks.map((x) => (x.id === id ? next : x)) };
   555→            }
   556→            return {};
   557→          });
   558→          break;
   559→      }
   560→    },
   561→
   562→    /* ---------------- lists ---------------- */

   564→      const existing = input.id ? get().lists.find((l) => l.id === input.id) : undefined;
