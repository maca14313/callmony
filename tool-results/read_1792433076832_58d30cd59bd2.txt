   145→   145→
   146→   146→export const useApp = create<AppState>((set, get) => {
   147→   147→  /* ---- internal helpers ---- */
   148→   148→  const touch = <T extends { updatedAt: number }>(e: T): T => ({ ...e, updatedAt: now() });
   149→   149→
   150→   150→  const persistContact = (c: Contact) => void idbPut("contacts", c);
   151→   151→  const persistEntry = (e: LedgerEntry) => void idbPut("entries", e);
   152→   152→  const persistList = (l: CalcList) => void idbPut("lists", l);
   153→   153→  const persistNote = (n: Note) => void idbPut("notes", n);
   154→   154→  const persistReminder = (r: Reminder) => void idbPut("reminders", r);
   155→   155→  const persistBank = (b: BankAccount) => void idbPut("banks", b);
   156→   156→  const persistRecent = (c: CallLog) => void idbPut("recents", c);
   157→   157→  const persistMessage = (m: MessageLog) => void idbPut("messages", m);
   158→   158→  const persistAttachment = (a: AttachmentMeta) => void idbPut("attachments", a);
   159→   159→  const persistSettings = (s: Settings) => void metaSet("settings", s);
   160→   160→
   161→   161→  const updateEntry = (id: ID, fn: (e: LedgerEntry) => LedgerEntry) => {
   162→   162→    const e = get().entries.find((x) => x.id === id);
   163→   163→    if (!e) return;
   164→   164→    const next = touch(fn(e));
   165→   165→    set((s) => ({ entries: s.entries.map((x) => (x.id === id ? next : x)) }));
   166→   166→    persistEntry(next);
   167→   167→  };
   168→   168→
   169→   169→  const updateList = (id: ID, fn: (l: CalcList) => CalcList) => {
   170→   170→    const l = get().lists.find((x) => x.id === id);
   171→   171→    if (!l) return;
   172→   172→    const next = touch(fn(l));
   173→   173→    set((s) => ({ lists: s.lists.map((x) => (x.id === id ? next : x)) }));
   174→   174→    persistList(next);
   175→   175→  };
   176→   176→
   177→   177→  const updateReminder = (id: ID, fn: (r: Reminder) => Reminder) => {
   178→   178→    const r = get().reminders.find((x) => x.id === id);
   179→   179→    if (!r) return;
   180→   180→    const next = touch(fn(r));
   181→   181→    set((s) => ({ reminders: s.reminders.map((x) => (x.id === id ? next : x)) }));
   182→   182→    persistReminder(next);
   183→   183→  };
   184→   184→
   185→   185→  return {
   186→   186→    hydrated: false,
   187→   187→    contacts: [],
   188→   188→    entries: [],
   189→   189→    lists: [],
   190→   190→    notes: [],
   191→   191→    reminders: [],
   192→   192→    banks: [],
   193→   193→    recents: [],
   194→   194→    messages: [],
   195→   195→    attachments: [],
   196→   196→    settings: DEFAULT_SETTINGS,
   197→   197→    lastUndo: null,
   198→   198→
   199→   199→    /* ---------------- hydrate ---------------- */
   200→   200→    hydrate: async () => {
   201→   201→      if (get().hydrated) return;
   202→   202→      const [contacts, entries, lists, notes, reminders, banks, recents, messages, attachments] =
   203→   203→        await Promise.all([
   204→   204→          idbGetAll<Contact>("contacts"),
   205→   205→          idbGetAll<LedgerEntry>("entries"),
   206→   206→          idbGetAll<CalcList>("lists"),
   207→   207→          idbGetAll<Note>("notes"),
   208→   208→          idbGetAll<Reminder>("reminders"),
   209→   209→          idbGetAll<BankAccount>("banks"),
   210→   210→          idbGetAll<CallLog>("recents"),
   211→   211→          idbGetAll<MessageLog>("messages"),
   212→   212→          idbGetAll<AttachmentMeta>("attachments"),
   213→   213→        ]);
   214→   214→      let settings = (await metaGet<Settings>("settings")) || DEFAULT_SETTINGS;
   215→   215→      settings = { ...DEFAULT_SETTINGS, ...settings };
   216→   216→
   217→   217→      const seeded = await metaGet<boolean>("demoSeeded");
   218→   218→      if (!seeded && contacts.length === 0 && entries.length === 0) {
   219→   219→        const demo = buildDemoData();
   220→   220→        set({
   221→   221→          contacts: demo.contacts,
   222→   222→          entries: demo.entries,
   223→   223→          lists: demo.lists,
   224→   224→          notes: demo.notes,
   225→   225→          reminders: demo.reminders,
   226→   226→          banks: demo.banks,
   227→   227→          recents: demo.recents,
   228→   228→          messages: [],
   229→   229→          attachments: [],
   230→   230→          settings: { ...settings, demoSeeded: true },
   231→   231→        });
   232→   232→        await metaSet("demoSeeded", true);
   233→   233→        demo.contacts.forEach(persistContact);
   234→   234→        demo.entries.forEach(persistEntry);
   235→   235→        demo.lists.forEach(persistList);
   236→   236→        demo.notes.forEach(persistNote);
   237→   237→        demo.reminders.forEach(persistReminder);
   238→   238→        demo.banks.forEach(persistBank);
   239→   239→        demo.recents.forEach(persistRecent);
   240→   240→        await metaSet("settings", { ...settings, demoSeeded: true });
   241→   241→        set({ hydrated: true });
   242→   242→        return;
   243→   243→      }
   244→   244→
   245→   245→      set({
   246→   246→        contacts,
   247→   247→        entries,
   248→   248→        lists,
   249→   249→        notes,
   250→   250→        reminders,
   251→   251→        banks,
   252→   252→        recents: recents.sort((a, b) => b.at - a.at),
   253→   253→        messages,
   254→   254→        attachments,
   255→   255→        settings,
   256→   256→        hydrated: true,
   257→   257→      });
   258→   258→    },
   259→   259→
   260→   260→    pushUndo: (label, fn) => set({ lastUndo: { label, fn } }),
   261→   261→    undo: () => {
   262→   262→      const u = get().lastUndo;
   263→   263→      if (!u) return false;
   264→   264→      set({ lastUndo: null });
   265→   265→      u.fn();
   266→   266→      return true;
   267→   267→    },
   268→   268→
   269→   269→    /* ---------------- contacts ---------------- */
   270→   270→    saveContact: (input) => {
   271→   271→      const existing = input.id ? get().contacts.find((c) => c.id === input.id) : undefined;
   272→   272→      const id = input.id || uid();
   273→   273→      const c: Contact = {
   274→   274→        id,
   275→   275→        firstName: input.firstName ?? existing?.firstName ?? "",
   276→   276→        lastName: input.lastName ?? existing?.lastName ?? "",
   277→   277→        nickname: input.nickname ?? existing?.nickname,
   278→   278→        company: input.company ?? existing?.company,
   279→   279→        jobTitle: input.jobTitle ?? existing?.jobTitle,
   280→   280→        phones: input.phones ?? existing?.phones ?? [],
   281→   281→        emails: input.emails ?? existing?.emails ?? [],
   282→   282→        links: input.links ?? existing?.links ?? [],
   283→   283→        socials: input.socials ?? existing?.socials ?? [],
   284→   284→        addresses: input.addresses ?? existing?.addresses ?? [],
   285→   285→        customFields: input.customFields ?? existing?.customFields ?? [],
   286→   286→        birthday: input.birthday ?? existing?.birthday,
   287→   287→        tags: input.tags ?? existing?.tags ?? [],
   288→   288→        favorite: input.favorite ?? existing?.favorite ?? false,
   289→   289→        pinnedNote: input.pinnedNote ?? existing?.pinnedNote,
   290→   290→        note: input.note ?? existing?.note,
   291→   291→        photoId: input.photoId ?? existing?.photoId,
   292→   292→        color: input.color ?? existing?.color,
   293→   293→        textColor: input.textColor ?? existing?.textColor,
   294→   294→        speedDial: input.speedDial ?? existing?.speedDial,
   295→   295→        createdAt: existing?.createdAt ?? now(),
   296→   296→        updatedAt: now(),
   297→   297→      };
   298→   298→      // only one primary phone/email
   299→   299→      if (c.phones.some((p) => p.primary)) c.phones.forEach((p, i) => { if (i > 0 && p.primary && c.phones[0].primary) p.primary = false; });
   300→   300→      set((s) => ({
   301→   301→        contacts: existing
   302→   302→          ? s.contacts.map((x) => (x.id === id ? c : x))
   303→   303→          : [...s.contacts, c],
   304→   304→      }));
   305→   305→      persistContact(c);
   306→   306→      return c;
   307→   307→    },
   308→   308→
   309→   309→    deleteContact: (id) => {
   310→   310→      const c = get().contacts.find((x) => x.id === id);
   311→   311→      if (!c) return;
   312→   312→      const next = { ...c, deletedAt: now() };
   313→   313→      set((s) => ({ contacts: s.contacts.map((x) => (x.id === id ? next : x)) }));
   314→   314→      persistContact(next);
   315→   315→      get().pushUndo(`“${displayName(c)}” moved to Trash`, () => get().restoreContact(id));
   316→   316→    },
   317→   317→
   318→   318→    restoreContact: (id) => {
   319→   319→      const c = get().contacts.find((x) => x.id === id);
   320→   320→      if (!c) return;
   321→   321→      const next = { ...c, deletedAt: undefined };
   322→   322→      set((s) => ({ contacts: s.contacts.map((x) => (x.id === id ? next : x)) }));
   323→   323→      persistContact(next);
   324→   324→    },
   325→   325→
   326→   326→    purgeContact: async (id) => {
   327→   327→      set((s) => ({ contacts: s.contacts.filter((x) => x.id !== id) }));
   328→   328→      await idbDel("contacts", id);
   329→   329→    },
   330→   330→
   331→   331→    toggleFavorite: (id) => {
   332→   332→      const c = get().contacts.find((x) => x.id === id);
   333→   333→      if (!c) return;
   334→   334→      const next = touch({ ...c, favorite: !c.favorite });
   335→   335→      set((s) => ({ contacts: s.contacts.map((x) => (x.id === id ? next : x)) }));
   336→   336→      persistContact(next);
   337→   337→    },
   338→   338→
   339→   339→    mergeContacts: (primaryId, dupId) => {
   340→   340→      const { contacts, entries, lists, notes, reminders, banks, recents, attachments } = get();
   341→   341→      const primary = contacts.find((c) => c.id === primaryId);
   342→   342→      const dup = contacts.find((c) => c.id === dupId);
   343→   343→      if (!primary || !dup) return;
   344→   344→
   345→   345→      const unionById = <T extends { id: string }>(a: T[], b: T[]) => {
   346→   346→        const ids = new Set(a.map((x) => x.id));
   347→   347→        return [...a, ...b.filter((x) => !ids.has(x.id))];
   348→   348→      };
   349→   349→      const phoneKeys = new Set(primary.phones.map((p) => normalizePhone(p.number)));
   350→   350→      const merged: Contact = touch({
   351→   351→        ...primary,
   352→   352→        phones: [
   353→   353→          ...primary.phones,
   354→   354→          ...dup.phones.filter((p) => !phoneKeys.has(normalizePhone(p.number))),
   355→   355→        ],
   356→   356→        emails: unionById(primary.emails, dup.emails),
   357→   357→        links: unionById(primary.links, dup.links),
   358→   358→        socials: unionById(primary.socials, dup.socials),
   359→   359→        addresses: unionById(primary.addresses, dup.addresses),
   360→   360→        customFields: unionById(primary.customFields, dup.customFields),
   361→   361→        tags: [...new Set([...primary.tags, ...dup.tags])],
   362→   362→        favorite: primary.favorite || dup.favorite,
   363→   363→        birthday: primary.birthday || dup.birthday,
   364→   364→        note: [primary.note, dup.note].filter(Boolean).join("\n") || undefined,
   365→   365→      });
   366→   366→
   367→   367→      const remapOwner = <T extends { ownerId: OwnerId }>(arr: T[], fn: (x: T) => void) =>
   368→   368→        arr.map((x) => {
   369→   369→          if (x.ownerId === dupId) {
   370→   370→            const n = { ...x, ownerId: primaryId, updatedAt: now() };
   371→   371→            fn(n as T);
   372→   372→            return n;
   373→   373→          }
   374→   374→          return x;
   375→   375→        });
   376→   376→
   377→   377→      const nextEntries = remapOwner(entries, (e) => persistEntry(e));
   378→   378→      const nextLists = remapOwner(lists, (l) => persistList(l));
   379→   379→      const nextNotes = remapOwner(notes, (n) => persistNote(n));
   380→   380→      const nextReminders = remapOwner(reminders, (r) => persistReminder(r));
   381→   381→      const nextBanks = remapOwner(banks, (b) => persistBank(b));
   382→   382→      const nextRecents = recents.map((r) =>
   383→   383→        r.contactId === dupId ? (persistRecent({ ...r, contactId: primaryId }), { ...r, contactId: primaryId }) : r
   384→   384→      );
   385→   385→      const nextAttachments = attachments.map((a) =>
   386→   386→        a.parentType === "contact" && a.parentId === dupId
   387→   387→          ? (persistAttachment({ ...a, parentId: primaryId }), { ...a, parentId: primaryId })
   388→   388→          : a
   389→   389→      );
   390→   390→
   391→   391→      set({
   392→   392→        contacts: contacts.map((x) => (x.id === primaryId ? merged : x)).filter((x) => x.id !== dupId),
   393→   393→        entries: nextEntries,
   394→   394→        lists: nextLists,
   395→   395→        notes: nextNotes,
   396→   396→        reminders: nextReminders,
   397→   397→        banks: nextBanks,
   398→   398→        recents: nextRecents,
   399→   399→        attachments: nextAttachments,
   400→   400→      });
   401→   401→      persistContact(merged);
   402→   402→      void idbDel("contacts", dupId);
   403→   403→    },
   404→   404→
   405→   405→    findDuplicates: () => {
   406→   406→      const { contacts } = get();
   407→   407→      const live = contacts.filter((c) => !c.deletedAt);
   408→   408→      const results: { primary: Contact; dup: Contact; reason: string }[] = [];
   409→   409→      const seen = new Set<ID>();
   410→   410→      for (let i = 0; i < live.length; i++) {
   411→   411→        for (let j = i + 1; j < live.length; j++) {
   412→   412→          const a = live[i];
   413→   413→          const b = live[j];
   414→   414→          if (seen.has(a.id) || seen.has(b.id)) continue;
   415→   415→          const aPhones = new Set(a.phones.map((p) => normalizePhone(p.number)));
   416→   416→          const sharedPhone = b.phones.some((p) => aPhones.has(normalizePhone(p.number)));
   417→   417→          const sameName =
   418→   418→            displayName(a).toLowerCase() === displayName(b).toLowerCase() && displayName(a) !== "Unknown";
   419→   419→          const aEmails = new Set(a.emails.map((e) => e.address.toLowerCase()));
   420→   420→          const sharedEmail = b.emails.some((e) => aEmails.has(e.address.toLowerCase()));
   421→   421→          if (sharedPhone || (sameName && sharedEmail)) {
   422→   422→            results.push({
   423→   423→              primary: a,
   424→   424→              dup: b,
   425→   425→              reason: sharedPhone ? "Same phone number" : "Same name and email",
   426→   426→            });
   427→   427→            seen.add(b.id);
   428→   428→          }
   429→   429→        }
   430→   430→      }
   431→   431→      return results;
   432→   432→    },

   508→   508→        .entries.filter((e) => e.ownerId === ownerId && !e.deletedAt)
   509→   509→        .sort((a, b) => a.date.localeCompare(b.date) || a.createdAt - b.createdAt);

   511→   511→        if (left < 0.005) break;
   512→   512→        const rem = calc.entryRemaining(e);
   513→   513→        // paying down: reduce |remaining| in the direction that zeroes the balance
   514→   514→        const effective = e.type === "credit" ? rem : -rem;
   515→   515→        if (Math.sign(effective) === Math.sign(balance)) continue; // same side, skip
   516→   516→        const pay = Math.min(Math.abs(rem), left);

   518→   518→          get().addPayment(e.id, { amount: r2Fallback(pay), date: todayStr(), auto: "settle", note: "Settle up" });
   519→   519→          left = r2Fallback(left - pay);
   520→   520→        }
   521→   521→      }
   522→   522→      return Math.abs(balance);

  1089→  1089→async function compressImage(
  1090→  1090→  file: File,
  1091→  1091→  maxSide: number,
  1092→  1092→  quality: number
  1093→  1093→): Promise<{ blob: Blob; width: number; height: number }> {
  1094→  1094→  const img = await new Promise<HTMLImageElement>((resolve, reject) => {
  1095→  1095→    const url = URL.createObjectURL(file);
  1096→  1096→    const i = new Image();
  1097→  1097→    i.onload = () => {
  1098→  1098→      URL.revokeObjectURL(url);
  1099→  1099→      resolve(i);
  1100→  1100→    };
  1101→  1101→    i.onerror = () => {
  1102→  1102→      URL.revokeObjectURL(url);
  1103→  1103→      reject(new Error("bad image"));
  1104→  1104→    };
  1105→  1105→    i.src = url;
  1106→  1106→  });
  1107→  1107→  const scale = Math.min(1, maxSide / Math.max(img.width, img.height));
  1108→  1108→  const w = Math.max(1, Math.round(img.width * scale));
  1109→  1109→  const h = Math.max(1, Math.round(img.height * scale));
  1110→  1110→  const canvas = document.createElement("canvas");
  1111→  1111→  canvas.width = w;
  1112→  1112→  canvas.height = h;
  1113→  1113→  const ctx = canvas.getContext("2d");
  1114→  1114→  if (!ctx) return { blob: file, width: img.width, height: img.height };
  1115→  1115→  ctx.drawImage(img, 0, 0, w, h);
  1116→  1116→  const blob = await new Promise<Blob>((resolve) =>
  1117→  1117→    canvas.toBlob((b) => resolve(b || file), "image/jpeg", quality)
  1118→  1118→  );
  1119→  1119→  return { blob, width: w, height: h };
  1120→  1120→}
