    66→const dueMs = (d: string) => new Date(`${d}T23:59:59`).getTime();

    68→const parseNum = (s: string): number => {
    69→  const n = parseFloat(String(s).replace(/,/g, ""));
    70→  return isNaN(n) ? 0 : n;

    73→/** Due-date chip — red when overdue, amber when today. */
    74→function DueBadge({ dueDate }: { dueDate: string }) {
    75→  const st = dueStatus(dueMs(dueDate));
    76→  return (
    77→    <span
    78→      className={cx(
    79→        "inline-flex shrink-0 items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-bold",
    80→        st === "overdue" ? "bg-negsoft text-neg" : st === "today" ? "bg-amber/15 text-amber" : "bg-surface2 text-mute"
    81→      )}
    82→    >
    83→      <CalendarClock size={11} aria-hidden />
    84→      {st === "overdue" ? "Overdue · " : st === "today" ? "Due today · " : "Due "}
    85→      {fdShort(dueDate)}
    86→    </span>
    87→  );
    88→}

   120→function AddPaymentSheet({
   121→  open,
   122→  onClose,
   123→  remaining,
   124→  onSubmit,
   125→}: {
   126→  open: boolean;
   127→  onClose: () => void;
   128→  remaining: number;
   129→  onSubmit: (p: { amount: number; date: string; note?: string; method?: string }) => void;
   130→}) {
   131→  const [amount, setAmount] = useState("");
   132→  const [date, setDate] = useState(todayStr());
   133→  const [note, setNote] = useState("");
   134→  const [method, setMethod] = useState("cash");
   135→  if (!open) return null;
   136→
   137→  const submit = () => {
   138→    const amt = parseNum(amount);
   139→    if (amt <= 0) {
   140→      toastErr("Enter an amount greater than 0");
   141→      return;
   142→    }
   143→    onSubmit({ amount: r2(amt), date: date || todayStr(), note: note.trim() || undefined, method });
   144→  };
   145→
   146→  return (
   147→    <BottomSheet open={open} onClose={onClose} title="Add payment">
   148→      <Field label="Amount">
   149→        <div className="flex w-full items-center gap-2">
   150→          <input
   151→            value={amount}
   152→            onChange={(e) => setAmount(e.target.value)}
   153→            inputMode="decimal"
   154→            placeholder="0"
   155→            aria-label="Payment amount"
   156→            className={cx(inputCls, "text-lg font-bold tabular-nums")}
   157→          />
   158→          <CalcButton onPick={(v) => setAmount(String(v))} />
   159→        </div>
   160→      </Field>
   161→      {remaining > 0.005 && (
   162→        <button
   163→          type="button"
   164→          onClick={() => setAmount(String(remaining))}
   165→          className="mt-2 flex items-center gap-1.5 rounded-full bg-brandsoft px-3 py-1.5 text-[11px] font-bold text-brand transition active:scale-95"
   166→          aria-label="Fill full remaining amount"
   167→        >
   168→          <HandCoins size={12} aria-hidden /> Full remaining: <Money value={remaining} />
   169→        </button>
   170→      )}
   171→      <div className="mt-3 grid grid-cols-2 gap-2">
   172→        <Field label="Date">
   173→          <input
   174→            type="date"
   175→            value={date}
   176→            onChange={(e) => setDate(e.target.value)}
   177→            aria-label="Payment date"
   178→            className={inputCls}
   179→          />
   180→        </Field>
   181→        <Field label="Method">
   182→          <select
   183→            value={method}
   184→            onChange={(e) => setMethod(e.target.value)}
   185→            aria-label="Payment method"
   186→            className={inputCls}
   187→          >
   188→            {PAYMENT_METHODS.map((m) => (
   189→              <option key={m} value={m}>
   190→                {m[0].toUpperCase() + m.slice(1)}
   191→              </option>
   192→            ))}
   193→          </select>
   194→        </Field>
   195→      </div>
   196→      <Field label="Note" className="mt-3">
   197→        <input
   198→          value={note}
   199→          onChange={(e) => setNote(e.target.value)}
   200→          placeholder="Optional note"
   201→          aria-label="Payment note"
   202→          className={inputCls}
   203→        />
   204→      </Field>
   205→      <Button variant="brand" size="lg" className="mt-4 w-full" onClick={submit}>
   206→        <Plus size={15} /> Add payment
   207→      </Button>
   208→    </BottomSheet>
   209→  );
   210→}

   216→function RemainingChip({ entry }: { entry: LedgerEntry }) {
   217→  const rem = entryRemaining(entry);
   218→  if (rem < -0.005) {
   219→    return (
   220→      <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-negsoft px-2 py-0.5 text-[10px] font-bold text-neg">
   221→        Overpaid by <Money value={-rem} />
   222→      </span>
   223→    );
   224→  }
   225→  if (Math.abs(rem) <= 0.005) {
   226→    return (
   227→      <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-brandsoft px-2 py-0.5 text-[10px] font-bold text-pos">
   228→        Settled ✓
   229→      </span>
   230→    );
   231→  }
   232→  return (
   233→    <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-surface2 px-2 py-0.5 text-[10px] font-bold text-mute">
   234→      Paid <Money value={r2(entry.amount - rem)} /> of <Money value={entry.amount} />
   235→    </span>
   236→  );
   237→}

   239→function EntryCard({
   240→  entry,
   241→  running,
   242→  isGeneral,
   243→  ownerId,
   244→  contact,
   245→}: {
   246→  entry: LedgerEntry;
   247→  running: number;
   248→  isGeneral: boolean;
   249→  ownerId: OwnerId;
   250→  contact: Contact | null;
   251→}) {
   252→  const ui = useUI();
   253→  const confirm = useConfirm();
   254→  const entries = useApp((s) => s.entries);
   255→  const settings = useApp((s) => s.settings);
   256→  const addPayment = useApp((s) => s.addPayment);
   257→  const removePayment = useApp((s) => s.removePayment);
   258→  const deleteEntry = useApp((s) => s.deleteEntry);
   259→  const moveEntity = useApp((s) => s.moveEntity);
   260→
   261→  const [expanded, setExpanded] = useState(false);
   262→  const [menuOpen, setMenuOpen] = useState(false);
   263→  const [payOpen, setPayOpen] = useState(false);
   264→  const [pickerOpen, setPickerOpen] = useState(false);
   265→  const [shareOpen, setShareOpen] = useState(false);
   266→
   267→  const credit = entry.type === "credit";
   268→  const remaining = entryRemaining(entry);
   269→  const paid = r2(entry.amount - remaining);
   270→  const phone = primaryPhone(contact);
   271→  const ownerEntries = useMemo(
   272→    () =>
   273→      entries
   274→        .filter((e) => e.ownerId === ownerId && !e.deletedAt)
   275→        .sort((a, b) => a.date.localeCompare(b.date) || a.createdAt - b.createdAt),
   276→    [entries, ownerId]
   277→  );
   278→  const ownerLabel = contact ? displayName(contact) : "General";
   279→
   280→  const removePay = async (p: Payment, idx: number) => {
   281→    const ok = await confirm({
   282→      title: `Remove payment ${idx + 1}?`,
   283→      message: `Payment of ${money(p.amount, settings.currency)} on ${fd(p.date)} will be removed.`,
   284→      confirmLabel: "Remove",
   285→      tone: "danger",
   286→    });
   287→    if (!ok) return;
   288→    removePayment(entry.id, p.id);
   289→    toastOk("Payment removed");
   290→  };
   291→
   292→  const onDelete = async () => {
   293→    const ok = await confirm({
   294→      title: "Delete entry?",
   295→      message: `“${entry.note || entry.category || (credit ? "Credit" : "Debit")}” and its payment history move to Trash.`,
   296→      confirmLabel: "Delete",
   297→      tone: "danger",
   298→    });
   299→    if (!ok) return;
   300→    setMenuOpen(false);
   301→    deleteEntry(entry.id);
   302→    toastUndoLast();
   303→  };
   304→
   305→  const menuActions: { key: string; icon: typeof Pencil; label: string; danger?: boolean; onClick: () => void }[] = [
   306→    {
   307→      key: "edit",
   308→      icon: Pencil,
   309→      label: "Edit",
   310→      onClick: () => {
   311→        setMenuOpen(false);
   312→        ui.open({ type: "entryEdit", entryId: entry.id, ownerId: entry.ownerId });
   313→      },
   314→    },
   315→    {
   316→      key: "pay",
   317→      icon: HandCoins,
   318→      label: "Add payment",
   319→      onClick: () => {
   320→        setMenuOpen(false);
   321→        setPayOpen(true);
   322→      },
   323→    },
   324→    {
   325→      key: "move",
   326→      icon: UserRound,
   327→      label: "Move to contact…",
   328→      onClick: () => {
   329→        setMenuOpen(false);
   330→        setPickerOpen(true);
   331→      },
   332→    },
   333→    ...(ownerId !== "general"
   334→      ? [
   335→          {
   336→            key: "general",
   337→            icon: Inbox,
   338→            label: "Move to General",
   339→            onClick: () => {
   340→              setMenuOpen(false);
   341→              moveEntity("entry", entry.id, "general");
   342→              toastOk("Moved to General");
   343→            },
   344→          },
   345→        ]
   346→      : []),
   347→    {
   348→      key: "share",
   349→      icon: Share2,
   350→      label: "Share statement…",
   351→      onClick: () => {
   352→        setMenuOpen(false);
   353→        setShareOpen(true);
   354→      },
   355→    },
   356→    { key: "delete", icon: Trash2, label: "Delete", danger: true, onClick: () => void onDelete() },
   357→  ];
   358→
   359→  const nPayments = entry.payments.length;
   360→
   361→  return (
   362→    <div className="rounded-[22px] border border-line bg-surface p-4 shadow-key">
   363→      {/* header */}
   364→      <div className="flex items-start gap-3">
   365→        <span
   366→          className={cx(
   367→            "flex h-10 w-10 shrink-0 items-center justify-center rounded-[14px]",
   368→            credit ? "bg-brandsoft text-pos" : "bg-negsoft text-neg"
   369→          )}
   370→          aria-hidden
   371→        >
   372→          {credit ? <ArrowDownLeft size={18} /> : <ArrowUpRight size={18} />}
   373→        </span>
   374→        <button
   375→          type="button"
   376→          className="min-w-0 flex-1 text-left"
   377→          onClick={() => setExpanded((v) => !v)}
   378→          aria-expanded={expanded}
   379→          aria-label={`Entry: ${entry.category || (credit ? "Credit" : "Debit")}. Tap to toggle payment history.`}
   380→        >
   381→          <span className="flex items-baseline justify-between gap-2">
   382→            <span className="truncate text-[13px] font-semibold text-ink">
   383→              {entry.category || (credit ? (isGeneral ? "Income" : "Credit") : isGeneral ? "Expense" : "Debit")}
   384→            </span>
   385→            <span className="shrink-0 text-[15px] font-bold">
   386→              <Money value={credit ? entry.amount : -entry.amount} signed />
   387→            </span>
   388→          </span>
   389→          <span className="mt-0.5 flex items-baseline justify-between gap-2">
   390→            <span className="truncate text-[11px] text-mute">
   391→              {fd(entry.date)}
   392→              {entry.method ? ` · ${entry.method}` : ""}
   393→            </span>
   394→            <span className="shrink-0 text-[10px] font-medium">
   395→              <Money value={running} signed className="text-mute" />
   396→            </span>
   397→          </span>
   398→        </button>
   399→        <IconButton
   400→          label="Entry actions"
   401→          className="h-10 w-10 shrink-0"
   402→          onClick={() => setMenuOpen(true)}
   403→        >
   404→          <Ellipsis size={16} />
   405→        </IconButton>
   406→      </div>
   407→
   408→      {/* meta chips + note */}
   409→      <div className="mt-2.5 flex flex-wrap items-center gap-1.5 pl-[52px]">
   410→        <RemainingChip entry={entry} />
   411→        {entry.dueDate && <DueBadge dueDate={entry.dueDate} />}
   412→        <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-surface2 px-2 py-0.5 text-[10px] font-semibold text-mute">
   413→          <ChevronDown
   414→            size={11}
   415→            className={cx("transition-transform", expanded && "rotate-180")}
   416→            aria-hidden
   417→          />
   418→          {nPayments} payment{nPayments !== 1 ? "s" : ""}
   419→        </span>
   420→      </div>
   421→      {entry.note && <p className="mt-1.5 pl-[52px] text-[11px] leading-4 text-mute">{entry.note}</p>}
   422→
   423→      {/* mini attachment strip — only mounted when there is something to show */}
   424→      {entry.attachments.length > 0 && (
   425→        <div className="mt-2 pl-[52px]">
   426→          <AttachmentStrip parentType="entry" parentId={entry.id} ids={entry.attachments} />
   427→        </div>
   428→      )}
   429→
   430→      {/* expandable payment history */}
   431→      <AnimatePresence initial={false}>
   432→        {expanded && (
   433→          <motion.div
   434→            initial={{ height: 0, opacity: 0 }}
   435→            animate={{ height: "auto", opacity: 1 }}
   436→            exit={{ height: 0, opacity: 0 }}
   437→            transition={{ duration: 0.2, ease: "easeOut" }}
   438→            className="overflow-hidden"
   439→          >
   440→            <div className="mt-3 border-t border-line pt-3">
   441→              <p className="mb-2 text-[10px] font-bold uppercase tracking-[1.5px] text-mute">Payment history</p>
   442→              {nPayments === 0 && <p className="pb-1 text-[11px] text-mute">No payments yet</p>}
   443→              <ol className="space-y-1.5">
   444→                {entry.payments.map((p, i) => (
   445→                  <li
   446→                    key={p.id}
   447→                    className="flex items-center gap-2 rounded-xl bg-surface2 px-2.5 py-1.5"
   448→                  >
   449→                    <span className="w-4 shrink-0 text-[10.5px] font-bold tabular-nums text-mute">{i + 1}.</span>
   450→                    <span className="shrink-0 text-[11px] text-mute">{fdShort(p.date)}</span>
   451→                    <span className="shrink-0 text-[12px] font-bold text-ink">
   452→                      <Money value={p.amount} />
   453→                    </span>
   454→                    {(p.note || p.method || p.auto) && (
   455→                      <span className="min-w-0 flex-1 truncate text-[10px] text-mute">
   456→                        {p.auto === "tick" ? "Ticked" : p.auto === "settle" ? "Settle up" : p.note || p.method}

   459→                    <IconButton

   462→                      onClick={() => void removePay(p, i)}
   463→                    >
   464→                      <X size={13} />

   469→              {entry.amount > 0 && nPayments > 0 && (
   470→                <ProgressBar className="mt-2.5" value={(paid / entry.amount) * 100} />
   471→              )}
   472→              <p className="mt-1.5 text-[10.5px] text-mute">
   473→                {nPayments} payment{nPayments !== 1 ? "s" : ""} ·{" "}
   474→                {remaining > 0.005 ? (
   475→                  <>
   476→                    <Money value={remaining} /> left
   477→                  </>
   478→                ) : remaining < -0.005 ? (
   479→                  "overpaid"
   480→                ) : (
   481→                  "all paid ✓"
   482→                )}
   483→              </p>
   484→              <div className="mt-2.5 flex gap-2">
   485→                <Button variant="soft" size="sm" onClick={() => setPayOpen(true)}>
   486→                  <Plus size={13} /> Add payment
   487→                </Button>
   488→                <Button
   489→                  variant="surface"
   490→                  size="sm"
   491→                  onClick={() => ui.open({ type: "entryEdit", entryId: entry.id, ownerId: entry.ownerId })}
   492→                >
   493→                  <Pencil size={13} /> Edit
   494→                </Button>
   495→              </div>
   496→            </div>
   497→          </motion.div>
   498→        )}
   499→      </AnimatePresence>
   500→
   501→      {/* ⋯ menu */}
   502→      <BottomSheet open={menuOpen} onClose={() => setMenuOpen(false)} title={entry.note || entry.category || "Entry"}>
   503→        <div className="space-y-1">
   504→          {menuActions.map((a) => {
   505→            const Icon = a.icon;
   506→            return (
   507→              <button
   508→                key={a.key}
   509→                type="button"
   510→                onClick={a.onClick}
   511→                className={cx(
   512→                  "flex w-full items-center gap-3 rounded-2xl px-3 py-3 text-left transition hover:bg-surface2 active:scale-[0.98]",
   513→                  a.danger && "text-neg"
   514→                )}
   515→              >
   516→                <Icon size={17} className={a.danger ? "text-neg" : "text-brand"} aria-hidden />
   517→                <span className="text-[13px] font-semibold">{a.label}</span>
   518→              </button>
   519→            );
   520→          })}
   521→        </div>
   522→      </BottomSheet>
   523→
   524→      {/* move to contact */}
   525→      <ContactPicker
   526→        open={pickerOpen}
   527→        onClose={() => setPickerOpen(false)}
   528→        onPick={(id) => {
   529→          moveEntity("entry", entry.id, id as OwnerId);
   530→          toastOk("Entry moved");
   531→        }}
   532→      />
   533→
   534→      {/* share statement */}
   535→      <BottomSheet open={shareOpen} onClose={() => setShareOpen(false)} title="Share statement">
   536→        <div className="space-y-1">
   537→          <button
   538→            type="button"
   539→            onClick={() => {
   540→              if (!phone) {
   541→                toastErr(contact ? "No phone number for this contact" : "General has no phone to send to");
   542→                return;
   543→              }
   544→              openExternal(waLink(phone, statementShareText(ownerEntries, contact, settings.currency)));
   545→            }}
   546→            className={cx(
   547→              "flex w-full items-center gap-3 rounded-2xl px-3 py-3 text-left transition hover:bg-surface2 active:scale-[0.98]",
   548→              !phone && "opacity-40"
   549→            )}
   550→          >
   551→            <MessageCircle size={17} className="text-brand" aria-hidden />
   552→            <span className="min-w-0 flex-1">
   553→              <span className="block text-[13px] font-semibold text-ink">WhatsApp</span>
   554→              <span className="block text-[10.5px] text-mute">
   555→                {phone ? `Send to ${ownerLabel}` : "No phone number available"}
   556→              </span>
   557→            </span>
   558→          </button>
   559→          <button
   560→            type="button"
   561→            onClick={() => {
   562→              downloadBlob(
   563→                statementToPdfBlob(ownerEntries, contact, settings.currency),
   564→                `${ownerLabel.replace(/[\\/:*?"<>|]+/g, "-")}-statement.pdf`
   565→              );
   566→              toastOk("Statement PDF downloaded");
   567→            }}
   568→            className="flex w-full items-center gap-3 rounded-2xl px-3 py-3 text-left transition hover:bg-surface2 active:scale-[0.98]"
   569→          >
   570→            <FileText size={17} className="text-brand" aria-hidden />
   571→            <span className="min-w-0 flex-1">
   572→              <span className="block text-[13px] font-semibold text-ink">PDF statement</span>
   573→              <span className="block text-[10.5px] text-mute">Download — entries with running balance</span>
   574→            </span>
   575→          </button>
   576→          <button
   577→            type="button"
   578→            onClick={async () => {
   579→              if (await copyText(statementShareText(ownerEntries, contact, settings.currency))) {
   580→                toastOk("Statement copied");
   581→              }
   582→            }}
   583→            className="flex w-full items-center gap-3 rounded-2xl px-3 py-3 text-left transition hover:bg-surface2 active:scale-[0.98]"
   584→          >
   585→            <Copy size={17} className="text-brand" aria-hidden />
   586→            <span className="min-w-0 flex-1">
   587→              <span className="block text-[13px] font-semibold text-ink">Copy text</span>
   588→              <span className="block text-[10.5px] text-mute">Paste it anywhere</span>
   589→            </span>
   590→          </button>

   592→            type="button"

   594→              const csv = entriesToCSV(ownerEntries, useApp.getState().contacts);
   595→              downloadBlob(
   596→                new Blob([csv], { type: "text/csv;charset=utf-8" }),
   597→                `${ownerLabel.replace(/[\\/:*?"<>|]+/g, "-")}-ledger.csv`

   600→            }}
   601→            className="flex w-full items-center gap-3 rounded-2xl px-3 py-3 text-left transition hover:bg-surface2 active:scale-[0.98]"

   603→            <FileDown size={17} className="text-brand" aria-hidden />

   619→          toastOk("Payment added");
   620→          setPayOpen(false);
   621→        }}
   622→      />
   623→    </div>
   624→  );

   631→export function LedgerSection({ ownerId }: { ownerId: OwnerId }) {
   632→  const ui = useUI();
   633→  const confirm = useConfirm();
   634→  const entries = useApp((s) => s.entries);
   635→  const lists = useApp((s) => s.lists);
   636→  const contacts = useApp((s) => s.contacts);
   637→  const settleUp = useApp((s) => s.settleUp);
   638→
   639→  const [filter, setFilter] = useState<LedgerFilter>("all");
   640→
   641→  const isGeneral = ownerId === "general";
   642→  const contact = useMemo(
   643→    () => (ownerId === "general" ? null : contacts.find((c) => c.id === ownerId && !c.deletedAt) ?? null),
   644→    [contacts, ownerId]
   645→  );
   646→  const ownerName = contact ? displayName(contact) : "General";
   647→
   648→  const totals = useMemo(() => totalsForOwner(entries, ownerId), [entries, ownerId]);
   649→  const listsPending = useMemo(() => openListsPending(lists, ownerId), [lists, ownerId]);
   650→
   651→  /** date desc rows with running balance computed oldest → newest */
   652→  const rows = useMemo(() => {
   653→    const mine = entries.filter((e) => e.ownerId === ownerId && !e.deletedAt);
   654→    const asc = [...mine].sort((a, b) => a.date.localeCompare(b.date) || a.createdAt - b.createdAt);
   655→    const running = new Map<ID, number>();
   656→    let run = 0;
   657→    for (const e of asc) {
   658→      run = r2(run + entrySignedRemaining(e));
   659→      running.set(e.id, run);
   660→    }
   661→    return asc.reverse().map((e) => ({ entry: e, running: running.get(e.id) ?? 0 }));
   662→  }, [entries, ownerId]);
   663→
   664→  const filtered = useMemo(() => {
   665→    if (filter === "all") return rows;
   666→    return rows.filter(({ entry }) => {
   667→      const rem = entryRemaining(entry);
   668→      if (filter === "credit") return entry.type === "credit";
   669→      if (filter === "debit") return entry.type === "debit";
   670→      if (filter === "unpaid") return rem > 0.005;
   671→      return rem > 0.005 && !!entry.dueDate && dueStatus(dueMs(entry.dueDate)) === "overdue";
   672→    });
   673→  }, [rows, filter]);
   674→
   675→  const filterLabels: Record<LedgerFilter, string> = {
   676→    all: "All",
   677→    credit: isGeneral ? "Income" : "Credit",
   678→    debit: isGeneral ? "Expense" : "Debit",
   679→    unpaid: "Unpaid",
   680→    overdue: "Overdue",
   681→  };
   682→
   683→  const openEntry = (entryType: "credit" | "debit") =>
   684→    ui.open({ type: "entryEdit", ownerId, entryType });
   685→
   686→  const onSettleUp = async () => {
   687→    const ok = await confirm({
   688→      title: "Settle up?",
   689→      message: isGeneral
   690→        ? "Adds payments dated today that bring the balance to zero."
   691→        : `Adds payments dated today that bring the balance with ${ownerName} to zero.`,
   692→      confirmLabel: "Settle up",
   693→      tone: "brand",
   694→    });
   695→    if (!ok) return;
   696→    const applied = settleUp(ownerId);
   697→    toastOk(applied > 0 ? "Settled ✓" : "Already settled up");
   698→  };
   699→
   700→  return (
   701→    <section aria-label={`Ledger for ${ownerName}`}>
   702→      <SectionLabel
   703→        right={
   704→          <span className="text-[10px] font-semibold text-mute">
   705→            {rows.length} entr{rows.length !== 1 ? "ies" : "y"}
   706→          </span>
   707→        }
   708→      >
   709→        {isGeneral ? "Income & Expense" : "Ledger"}
   710→      </SectionLabel>
   711→
   712→      {/* summary strip */}
   713→      <div className="grid grid-cols-3 gap-2">
   714→        <StatCard label={isGeneral ? "Income" : "They owe you"} value={totals.credit} tone="pos" />
   715→        <StatCard label={isGeneral ? "Expense" : "You owe them"} value={totals.debit} tone="neg" />
   716→        <StatCard
   717→          label="Net"
   718→          value={totals.net}
   719→          tone={totals.net > 0.005 ? "pos" : totals.net < -0.005 ? "neg" : "mute"}
   720→          signed
   721→        />
   722→      </div>
   723→
   724→      {listsPending !== 0 && (
   725→        <p className="mt-2 flex items-center gap-1.5 rounded-2xl bg-amber/10 px-3 py-2 text-[11px] font-semibold text-amber">
   726→          <CalendarClock size={13} className="shrink-0" aria-hidden />
   727→          <span>
   728→            Open calculating lists: +<Money value={listsPending} /> (counted in balance)
   729→          </span>
   730→        </p>
   731→      )}
   732→
   733→      {/* actions */}
   734→      <div className="mt-3 flex flex-wrap gap-2">
   735→        <Button variant="surface" size="sm" onClick={() => void onSettleUp()}>
   736→          <Scale size={13} /> Settle up
   737→        </Button>
   738→        <Button variant="soft" size="sm" className="text-pos" onClick={() => openEntry("credit")}>
   739→          <ArrowDownLeft size={13} /> {isGeneral ? "+ Income" : "+ Credit"}
   740→        </Button>
   741→        <Button variant="surface" size="sm" className="text-neg" onClick={() => openEntry("debit")}>
   742→          <ArrowUpRight size={13} /> {isGeneral ? "+ Expense" : "+ Debit"}
   743→        </Button>
   744→      </div>
   745→
   746→      {/* filters */}
   747→      <div className="app-scroll mt-3 flex gap-1.5 overflow-x-auto pb-1" role="group" aria-label="Ledger filters">
   748→        {(Object.keys(filterLabels) as LedgerFilter[]).map((f) => (
   749→          <Chip key={f} active={filter === f} onClick={() => setFilter(f)}>
   750→            {filterLabels[f]}
   751→          </Chip>
   752→        ))}
   753→      </div>
   754→
   755→      {/* entries */}
   756→      <div className="mt-3 space-y-2.5">
   757→        {filtered.map(({ entry, running }) => (
   758→          <EntryCard
   759→            key={entry.id}
   760→            entry={entry}
   761→            running={running}
   762→            isGeneral={isGeneral}
   763→            ownerId={ownerId}
   764→            contact={contact}
   765→          />
   766→        ))}
   767→
   768→        {!filtered.length && rows.length > 0 && (
   769→          <EmptyState
   770→            icon={<Banknote size={20} />}
   771→            title="Nothing matches this filter"
   772→            hint="Try another filter to see more entries."
   773→            action={
   774→              <Button variant="soft" size="sm" onClick={() => setFilter("all")}>
   775→                Show all
   776→              </Button>
   777→            }
   778→          />
   779→        )}
   780→
   781→        {!rows.length && (
   782→          <EmptyState
   783→            icon={<Banknote size={22} />}
   784→            title={isGeneral ? "No income or expenses yet" : "No ledger entries yet"}
   785→            hint={
   786→              isGeneral
   787→                ? "Track money in and out — everything stays on this device."
   788→                : `Track what you and ${ownerName} owe each other. Settle up anytime.`
   789→            }
   790→            action={
   791→              <div className="flex gap-2">
   792→                <Button variant="soft" size="sm" className="text-pos" onClick={() => openEntry("credit")}>
   793→                  <Plus size={13} /> {isGeneral ? "Income" : "Credit"}
   794→                </Button>
   795→                <Button variant="surface" size="sm" className="text-neg" onClick={() => openEntry("debit")}>
   796→                  <Plus size={13} /> {isGeneral ? "Expense" : "Debit"}
   797→                </Button>
   798→              </div>
   799→            }
   800→          />
   801→        )}
   802→      </div>
   803→    </section>
   804→  );
   805→}
