    79→const SPEED_KEYS = ["2", "3", "4", "5", "6", "7", "8", "9"];
    80→
    81→/**
    82→ * Speed-dial assignment per key. `settings.speedDial` wins; contacts that
    83→ * still carry a legacy `speedDial` field act as a fallback (demo data).
    84→ */
    85→function speedDialAssignments(contacts: Contact[], settings: Settings): Record<string, string | undefined> {
    86→  const out: Record<string, string | undefined> = {};
    87→  contacts.forEach((c) => {
    88→    if (c.speedDial) out[c.speedDial] = c.id;
    89→  });
    90→  const sd = (settings as Settings & { speedDial?: Record<string, string | undefined> }).speedDial ?? {};
    91→  SPEED_KEYS.forEach((k) => {
    92→    if (k in sd) out[k] = sd[k];
    93→  });
    94→  return out;
    95→}

    97→/** Share a contact as a vCard — Web Share when available, copy / download otherwise. */
    98→async function shareVCard(c: Contact) {
    99→  const vcf = toVCard([c], true);
   100→  const res = await shareText(`Contact · ${displayName(c)}`, vcf);
   101→  if (res === "copied") {
   102→    await copyText(vcf);
   103→    toastOk("vCard copied to clipboard");
   104→  } else if (res === "failed") {
   105→    downloadBlob(new Blob([vcf], { type: "text/vcard" }), `${displayName(c)}.vcf`);
   106→    toastOk("vCard downloaded");
   107→  }
   108→}

   179→function ContactRow({
   180→  contact,
   181→  balance,
   182→  blocked,
   183→  onOpen,
   184→  onCall,
   185→  onFavorite,
   186→  onWhatsApp,
   187→  onMessage,
   188→  onDelete,
   189→  onLongPress,
   190→}: {
   191→  contact: Contact;
   192→  balance: number;
   193→  blocked: boolean;
   194→  onOpen: () => void;
   195→  onCall: () => void;
   196→  onFavorite: () => void;
   197→  onWhatsApp: () => void;
   198→  onMessage: () => void;
   199→  onDelete: () => void;
   200→  onLongPress: () => void;
   201→}) {
   202→  const phone = primaryPhone(contact);
   203→  const sub = phone
   204→    ? prettyPhone(phone)
   205→    : contact.socials.some((s) => s.app === "telegram")
   206→      ? "Telegram only"
   207→      : contact.emails[0]?.address || "No phone number";
   208→
   209→  const left: SwipeAction[] = [
   210→    { key: "call", icon: <Phone size={18} />, label: "Call", className: "bg-brand", onClick: onCall },
   211→    {
   212→      key: "fav",
   213→      icon: <Star size={18} fill={contact.favorite ? "currentColor" : "none"} />,
   214→      label: contact.favorite ? "Unfavorite" : "Favorite",
   215→      className: "bg-brandink",
   216→      onClick: onFavorite,
   217→    },
   218→  ];
   219→  const right: SwipeAction[] = [
   220→    {
   221→      key: "wa",
   222→      icon: <MessageCircle size={18} />,
   223→      label: "WhatsApp",
   224→      className: "bg-[#1faa53]",
   225→      onClick: onWhatsApp,
   226→    },
   227→    { key: "sms", icon: <Send size={18} />, label: "Message", className: "bg-brandink", onClick: onMessage },
   228→    { key: "del", icon: <Trash2 size={18} />, label: "Delete contact", className: "bg-neg", onClick: onDelete },
   229→  ];
   230→
   231→  return (
   232→    <SwipeRow className="mb-1.5" left={left} right={right} onTap={onOpen} onLongPress={onLongPress}>
   233→      <div className="flex items-center gap-3 rounded-2xl bg-surface px-3.5 py-2.5 shadow-key">
   234→        <Avatar contact={contact} size="md" />
   235→        <div className="min-w-0 flex-1">
   236→          <div className="flex items-center gap-1.5">
   237→            <p className="truncate text-[13px] font-semibold text-ink">{displayName(contact)}</p>
   238→            {contact.favorite && (
   239→              <Star size={11} className="shrink-0 text-brand" fill="currentColor" aria-label="Favorite" />
   240→            )}
   241→            {blocked && (
   242→              <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-negsoft px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wide text-neg">
   243→                <Ban size={9} /> blocked
   244→              </span>
   245→            )}
   246→          </div>
   247→          <p className="truncate text-[11px] text-mute">{sub}</p>
   248→        </div>
   249→        <div className="flex shrink-0 items-center gap-1">
   250→          <BalancePill value={balance} size="sm" />
   251→          <ChevronRight size={14} className="text-line2" />
   252→        </div>
   253→      </div>
   254→    </SwipeRow>
   255→  );
   256→}

   258→export function ContactsView() {
   259→  const app = useApp();
   260→  const ui = useUI();
   261→  const confirm = useConfirm();
   262→  const [q, setQ] = useState("");
   263→  const [filter, setFilter] = useState<FilterKey>("all");
   264→  const [sort, setSort] = useState<SortKey>("name");
   265→  const [menuFor, setMenuFor] = useState<Contact | null>(null);
   266→
   267→  const cc = app.settings.countryCode;
   268→  const contacts = useMemo(() => app.contacts.filter((c) => !c.deletedAt), [app.contacts]);
   269→
   270→  const balances = useMemo(() => {
   271→    const map = new Map<string, number>();
   272→    contacts.forEach((c) => map.set(c.id, ownerFullBalance(app.entries, app.lists, c.id)));
   273→    return map;
   274→  }, [contacts, app.entries, app.lists]);
   275→
   276→  const lastActivity = useMemo(() => {
   277→    const map = new Map<string, number>();
   278→    contacts.forEach((c) => map.set(c.id, c.createdAt));
   279→    app.entries.forEach((e) => {
   280→      if (!e.deletedAt) map.set(e.ownerId, Math.max(map.get(e.ownerId) ?? 0, e.updatedAt));
   281→    });
   282→    app.lists.forEach((l) => {
   283→      if (!l.deletedAt) map.set(l.ownerId, Math.max(map.get(l.ownerId) ?? 0, l.updatedAt));
   284→    });
   285→    app.recents.forEach((r) => {
   286→      if (r.contactId) map.set(r.contactId, Math.max(map.get(r.contactId) ?? 0, r.at));
   287→    });
   288→    return map;
   289→  }, [contacts, app.entries, app.lists, app.recents]);
   290→
   291→  const { toReceive, toPay, net } = useMemo(() => {
   292→    let pos = 0;
   293→    let neg = 0;
   294→    contacts.forEach((c) => {
   295→      const b = balances.get(c.id) ?? 0;
   296→      if (b > 0) pos += b;
   297→      if (b < 0) neg += b;
   298→    });
   299→    return { toReceive: r2(pos), toPay: r2(neg), net: r2(pos + neg) };
   300→  }, [contacts, balances]);
   301→
   302→  const blockedSet = useMemo(() => new Set(app.settings.blockedNumbers), [app.settings.blockedNumbers]);
   303→  const isBlocked = (c: Contact) => c.phones.some((p) => blockedSet.has(normalizePhone(p.number, cc)));
   304→
   305→  const filtered = useMemo(() => {
   306→    const query = q.toLowerCase().trim();
   307→    const digits = query.replace(/[^\d]/g, "");
   308→    const list = contacts.filter((c) => {
   309→      const b = balances.get(c.id) ?? 0;
   310→      if (filter === "owe-me" && b <= 0.005) return false;
   311→      if (filter === "i-owe" && b >= -0.005) return false;
   312→      if (filter === "settled" && Math.abs(b) > 0.005) return false;
   313→      if (filter === "favorites" && !c.favorite) return false;
   314→      if (!query) return true;
   315→      const hay = [
   316→        displayName(c),
   317→        c.nickname || "",
   318→        c.company || "",
   319→        c.jobTitle || "",
   320→        c.note || "",
   321→        c.pinnedNote || "",
   322→        ...c.tags,
   323→        ...c.phones.map((p) => p.number),
   324→        ...c.emails.map((e) => e.address),
   325→        ...c.socials.map((s) => s.handle),
   326→      ]
   327→        .join(" ")
   328→        .toLowerCase();
   329→      const numberHit =
   330→        digits.length >= 2 && c.phones.some((p) => p.number.replace(/\D/g, "").includes(digits));
   331→      return hay.includes(query) || numberHit;
   332→    });
   333→    return [...list].sort((a, b) => {
   334→      if (sort === "balance")
   335→        return (balances.get(b.id) ?? 0) - (balances.get(a.id) ?? 0) || displayName(a).localeCompare(displayName(b));
   336→      if (sort === "recent")
   337→        return (
   338→          (lastActivity.get(b.id) ?? 0) - (lastActivity.get(a.id) ?? 0) ||
   339→          displayName(a).localeCompare(displayName(b))
   340→        );
   341→      return displayName(a).localeCompare(displayName(b));
   342→    });
   343→  }, [contacts, q, filter, sort, balances, lastActivity]);
   344→
   345→  const groups = useMemo(() => {
   346→    if (sort !== "name") return [{ letter: "", items: filtered }];
   347→    const out: { letter: string; items: Contact[] }[] = [];
   348→    filtered.forEach((c) => {
   349→      const ch = (displayName(c)[0] || "#").toUpperCase();
   350→      const letter = /[A-Z]/.test(ch) ? ch : "#";
   351→      const last = out[out.length - 1];
   352→      if (last && last.letter === letter) last.items.push(c);
   353→      else out.push({ letter, items: [c] });
   354→    });
   355→    return out;
   356→  }, [filtered, sort]);
   357→
   358→  /* ---- actions ---- */
   359→  const call = (c: Contact) => {
   360→    const p = primaryPhone(c);
   361→    if (!p) {
   362→      toastErr("No phone number for this contact");
   363→      return;
   364→    }
   365→    ui.open({ type: "call", phone: p, contactId: c.id });
   366→  };
   367→  const whatsApp = (c: Contact) => {
   368→    const p = primaryPhone(c);
   369→    if (!p) {
   370→      toastErr("No phone number on this contact");
   371→      return;
   372→    }
   373→    openExternal(waLink(p, undefined, cc));
   374→  };
   375→  const message = (c: Contact) => {
   376→    const p = primaryPhone(c);
   377→    if (!p) {
   378→      toastErr("No phone number on this contact");
   379→      return;
   380→    }
   381→    openExternal(smsLink(p, undefined, cc));
   382→  };
   383→  const remove = async (c: Contact) => {
   384→    const ok = await confirm({
   385→      title: "Delete contact?",
   386→      message: `“${displayName(c)}” will move to Trash. You can undo right after, and ledger history stays safe.`,
   387→      confirmLabel: "Delete",
   388→      tone: "danger",
   389→    });
   390→    if (!ok) return;
   391→    app.deleteContact(c.id);
   392→    toastUndoLast();
   393→  };
   394→  const toggleBlock = (c: Contact) => {
   395→    const nums = c.phones.map((p) => normalizePhone(p.number, cc));
   396→    if (!nums.length) {
   397→      toastErr("No number to block");
   398→      return;
   399→    }
   400→    const blocked = nums.every((n) => blockedSet.has(n));
   401→    const next = blocked
   402→      ? app.settings.blockedNumbers.filter((b) => !nums.includes(b))
   403→      : [...new Set([...app.settings.blockedNumbers, ...nums])];
   404→    app.updateSettings({ blockedNumbers: next });
   405→    setMenuFor(null);
   406→    toastOk(blocked ? "Number unblocked" : "Number blocked");
   407→  };
   408→  const cycleSort = () =>
   409→    setSort((s) => (s === "name" ? "balance" : s === "balance" ? "recent" : "name"));
   410→
   411→  const menu = menuFor;
   412→  const menuPhone = menu ? primaryPhone(menu) : undefined;
   413→  const menuTg = menu ? telegramHandle(menu) : undefined;
   414→
   415→  return (
   416→    <div className="tab-content">
   417→      <ViewHeader title="Contacts" subtitle="The people who matter." />
   418→
   419→      <SearchInput value={q} onChange={setQ} placeholder="Search name, number, notes…" />
   420→
   421→      {/* summary strip — only when balances exist */}
   422→      {(toReceive > 0.005 || toPay < -0.005) && (
   423→        <div className="mt-3 grid grid-cols-3 gap-2">
   424→          <div className="rounded-2xl border border-line bg-surface p-3 shadow-key">
   425→            <p className="text-[9px] font-bold uppercase tracking-[1.5px] text-mute">To receive</p>
   426→            <p className="mt-1 text-[13px] font-bold text-pos">
   427→              <Money value={toReceive} signed />
   428→            </p>
   429→          </div>
   430→          <div className="rounded-2xl border border-line bg-surface p-3 shadow-key">
   431→            <p className="text-[9px] font-bold uppercase tracking-[1.5px] text-mute">To pay</p>
   432→            <p className="mt-1 text-[13px] font-bold text-neg">
   433→              <Money value={toPay} signed />
   434→            </p>
   435→          </div>
   436→          <div className="rounded-2xl border border-line bg-surface p-3 shadow-key">
   437→            <p className="text-[9px] font-bold uppercase tracking-[1.5px] text-mute">Net</p>
   438→            <p
   439→              className={cx(
   440→                "mt-1 text-[13px] font-bold",
   441→                net > 0.005 ? "text-pos" : net < -0.005 ? "text-neg" : "text-ink"
   442→              )}
   443→            >
   444→              <Money value={net} signed />
   445→            </p>
   446→          </div>
   447→        </div>
   448→      )}
   449→
   450→      {/* filter chips */}
   451→      <div className="app-scroll -mx-1 mt-3 flex gap-1.5 overflow-x-auto px-1 pb-1">
   452→        <Chip className="py-2" active={filter === "all"} onClick={() => setFilter("all")}>
   453→          All
   454→        </Chip>
   455→        <Chip className="py-2" active={filter === "owe-me"} onClick={() => setFilter("owe-me")}>
   456→          They owe me
   457→        </Chip>
   458→        <Chip className="py-2" active={filter === "i-owe"} onClick={() => setFilter("i-owe")}>
   459→          I owe them
   460→        </Chip>
   461→        <Chip className="py-2" active={filter === "settled"} onClick={() => setFilter("settled")}>
   462→          Settled
   463→        </Chip>
   464→        <Chip className="py-2" active={filter === "favorites"} onClick={() => setFilter("favorites")}>
   465→          Favorites
   466→        </Chip>
   467→      </div>
   468→
   469→      <div className="mt-4">
   470→        <SectionLabel
   471→          right={
   472→            <div className="flex items-center gap-2">
   473→              <button
   474→                type="button"
   475→                onClick={cycleSort}
   476→                className="flex h-11 items-center gap-1.5 rounded-full border border-line2 bg-surface px-3.5 text-[10px] font-bold uppercase tracking-[1.2px] text-mute shadow-key transition active:scale-95"
   477→                aria-label={`Sort by ${SORT_LABEL[sort]} — tap to change`}
   478→              >
   479→                <ArrowDownUp size={13} className="text-brand" />
   480→                {SORT_LABEL[sort]}
   481→              </button>
   482→              <button
   483→                type="button"
   484→                onClick={() => ui.open({ type: "contactEdit" })}
   485→                className="flex h-11 items-center gap-1 rounded-full bg-brandsoft px-3.5 text-[11px] font-bold text-brand transition active:scale-95"
   486→                aria-label="Add new contact"
   487→              >
   488→                <Plus size={14} /> Add new
   489→              </button>
   490→            </div>
   491→          }
   492→        >
   493→          {filtered.length === contacts.length
   494→            ? `${contacts.length} contact${contacts.length === 1 ? "" : "s"}`
   495→            : `${filtered.length} of ${contacts.length}`}
   496→        </SectionLabel>
   497→
   498→        {filtered.length === 0 ? (
   499→          <EmptyState
   500→            icon={<Users size={22} />}
   501→            title={q || filter !== "all" ? "No matching contacts" : "No contacts yet"}
   502→            hint={
   503→              q || filter !== "all"
   504→                ? "Try a different search or filter."
   505→                : "Add the people you call and settle with most."
   506→            }
   507→            action={
   508→              contacts.length === 0 ? (
   509→                <Button variant="brand" onClick={() => ui.open({ type: "contactEdit" })}>
   510→                  <Plus size={15} /> Add contact
   511→                </Button>
   512→              ) : undefined
   513→            }
   514→          />
   515→        ) : (
   516→          <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.22 }}>
   517→            {groups.map((g) => (
   518→              <div key={g.letter || "all"}>
   519→                {g.letter && (
   520→                  <div className="sticky top-0 z-10 bg-canvas/95 px-1 pb-1.5 pt-2 backdrop-blur-sm">
   521→                    <p className="text-[11px] font-bold uppercase tracking-[2px] text-brand">{g.letter}</p>
   522→                  </div>
   523→                )}
   524→                {g.items.map((c) => (
   525→                  <ContactRow
   526→                    key={c.id}
   527→                    contact={c}
   528→                    balance={balances.get(c.id) ?? 0}
   529→                    blocked={isBlocked(c)}
   530→                    onOpen={() => ui.open({ type: "contact", contactId: c.id })}
   531→                    onCall={() => call(c)}
   532→                    onFavorite={() => {
   533→                      app.toggleFavorite(c.id);
   534→                      toastOk(c.favorite ? "Removed from favorites" : "Added to favorites");
   535→                    }}
   536→                    onWhatsApp={() => whatsApp(c)}
   537→                    onMessage={() => message(c)}
   538→                    onDelete={() => void remove(c)}
   539→                    onLongPress={() => setMenuFor(c)}
   540→                  />
   541→                ))}
   542→              </div>
   543→            ))}
   544→          </motion.div>
   545→        )}
   546→      </div>
   547→
   548→      {/* long-press context menu */}
   549→      <BottomSheet
   550→        open={!!menu}
   551→        onClose={() => setMenuFor(null)}
   552→        title={menu ? displayName(menu) : ""}

   844→

   846→    const byId = new Map(contacts.map((c) => [c.id, c] as const));
   847→    const byPhone = new Map<string, Contact>();
   848→    contacts.forEach((c) =>
   849→      c.phones.forEach((p) => byPhone.set(normalizePhone(p.number, cc), c))
   850→    );

   855→

   857→    const query = q.toLowerCase().trim();
   858→    const digits = query.replace(/[^\d]/g, "");
   859→    const list = recents.filter((r) => {
   860→      if (dir !== "all" && r.direction !== dir) return false;
   861→      if (!query) return true;
   862→      const c = contactFor(r);
   863→      const hay = `${c ? displayName(c) : ""} ${r.phone}`.toLowerCase();
   864→      return hay.includes(query) || (digits.length >= 2 && r.phone.replace(/\D/g, "").includes(digits));
