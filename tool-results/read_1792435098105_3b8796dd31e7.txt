   209→/** Numeric input with local draft state — commits on blur, CalcButton optional. */
   210→function NumField({
   211→  value,
   212→  onCommit,
   213→  ariaLabel,
   214→  placeholder,
   215→  calc,
   216→  className,
   217→}: {
   218→  value: number;
   219→  onCommit: (n: number) => void;
   220→  ariaLabel: string;
   221→  placeholder?: string;
   222→  calc?: boolean;
   223→  className?: string;
   224→}) {
   225→  const [text, setText] = useState(value ? String(value) : "");
   226→  return (
   227→    <div className="flex w-full items-center gap-2">
   228→      <input
   229→        value={text}
   230→        inputMode="decimal"
   231→        placeholder={placeholder || "0"}
   232→        aria-label={ariaLabel}
   233→        onChange={(e) => setText(e.target.value)}
   234→        onBlur={() => onCommit(parseNum(text))}
   235→        className={cx(inputCls, "py-2.5 text-[13px] font-semibold tabular-nums", className)}
   236→      />
   237→      {calc && (
   238→        <CalcButton
   239→          onPick={(v) => {
   240→            setText(v ? String(v) : "");
   241→            onCommit(r2(v));
   242→          }}
   243→        />
   244→      )}
   245→    </div>
   246→  );

   304→/** Add-payment bottom sheet — shared by entry + item payment histories. */
   305→function AddPaymentSheet({
   306→  open,
   307→  onClose,
   308→  remaining,
   309→  currency,
   310→  onSubmit,
   311→}: {
   312→  open: boolean;
   313→  onClose: () => void;
   314→  remaining: number;
   315→  currency: string;
   316→  onSubmit: (p: { amount: number; date: string; note?: string; method?: string }) => void;
   317→}) {
   318→  const [amount, setAmount] = useState("");
   319→  const [date, setDate] = useState(todayStr());
   320→  const [note, setNote] = useState("");
   321→  const [method, setMethod] = useState("cash");
   322→  if (!open) return null;
   323→
   324→  const submit = () => {
   325→    const amt = parseNum(amount);
   326→    if (amt <= 0) {
   327→      toastErr("Enter an amount greater than 0");
   328→      return;
   329→    }
   330→    onSubmit({ amount: r2(amt), date: date || todayStr(), note: note.trim() || undefined, method });
   331→  };
   332→
   333→  return (
   334→    <BottomSheet open={open} onClose={onClose} title="Add payment">
   335→      <Field label="Amount">
   336→        <div className="flex w-full items-center gap-2">
   337→          <input
   338→            value={amount}
   339→            onChange={(e) => setAmount(e.target.value)}
   340→            inputMode="decimal"
   341→            placeholder="0"
   342→            aria-label="Payment amount"
   343→            className={cx(inputCls, "text-lg font-bold tabular-nums")}
   344→          />
   345→          <CalcButton onPick={(v) => setAmount(String(v))} />
   346→        </div>
   347→      </Field>
   348→      {remaining > 0.005 && (
   349→        <button
   350→          type="button"
   351→          onClick={() => setAmount(String(remaining))}
   352→          className="mt-2 flex items-center gap-1.5 rounded-full bg-brandsoft px-3 py-1.5 text-[11px] font-bold text-brand transition active:scale-95"
   353→          aria-label="Fill full remaining amount"
   354→        >
   355→          <HandCoins size={12} aria-hidden /> Full remaining: <Money value={remaining} currency={currency} />
   356→        </button>
   357→      )}
   358→      <div className="mt-3 grid grid-cols-2 gap-2">
   359→        <Field label="Date">
   360→          <input
   361→            type="date"
   362→            value={date}
   363→            onChange={(e) => setDate(e.target.value)}
   364→            aria-label="Payment date"
   365→            className={inputCls}
   366→          />
   367→        </Field>
   368→        <Field label="Method">
   369→          <select
   370→            value={method}
   371→            onChange={(e) => setMethod(e.target.value)}
   372→            aria-label="Payment method"
   373→            className={inputCls}
   374→          >
   375→            {PAYMENT_METHODS.map((m) => (
   376→              <option key={m} value={m}>
   377→                {m[0].toUpperCase() + m.slice(1)}
   378→              </option>
   379→            ))}
   380→          </select>
   381→        </Field>
   382→      </div>
   383→      <Field label="Note" className="mt-3">
   384→        <input
   385→          value={note}
   386→          onChange={(e) => setNote(e.target.value)}
   387→          placeholder="Optional note"
   388→          aria-label="Payment note"
   389→          className={inputCls}
   390→        />
   391→      </Field>
   392→      <Button variant="brand" size="lg" className="mt-4 w-full" onClick={submit}>
   393→        <Plus size={15} /> Add payment
   394→      </Button>
   395→    </BottomSheet>
   396→  );
   397→}

   403→function ItemPayments({ listId, item, currency }: { listId: ID; item: ListItem; currency: string }) {
   404→  const confirm = useConfirm();
   405→  const addItemPayment = useApp((s) => s.addItemPayment);
   406→  const removeItemPayment = useApp((s) => s.removeItemPayment);
   407→  const [payOpen, setPayOpen] = useState(false);
   408→
   409→  const lt = lineTotal(item);
   410→  const paid = itemPaid(item);
   411→  const rem = itemRemaining(item);
   412→  const n = item.payments.length;
   413→
   414→  const remove = async (p: Payment, idx: number) => {
   415→    const ok = await confirm({
   416→      title: `Remove payment ${idx + 1}?`,
   417→      message: `Payment of ${money(p.amount, currency)} on ${fd(p.date)} will be removed.`,
   418→      confirmLabel: "Remove",
   419→      tone: "danger",
   420→    });
   421→    if (!ok) return;
   422→    removeItemPayment(listId, item.id, p.id);
   423→    toastOk("Payment removed");
   424→  };
   425→
   426→  return (
   427→    <div className="rounded-2xl bg-surface2 p-3">
   428→      <p className="mb-2 text-[10px] font-bold uppercase tracking-[1.5px] text-mute">Payments</p>
   429→      {n === 0 && <p className="pb-1 text-[11px] text-mute">No payments yet</p>}
   430→      <ol className="space-y-1">
   431→        {item.payments.map((p, i) => (
   432→          <li key={p.id} className="flex items-center gap-2 rounded-xl bg-surface px-2.5 py-1.5 shadow-key">
   433→            <span className="w-3.5 shrink-0 text-[10.5px] font-bold tabular-nums text-mute">{i + 1}.</span>
   434→            <span className="shrink-0 text-[11px] text-mute">{fdShort(p.date)}</span>
   435→            <span className="shrink-0 text-[12px] font-bold text-ink">
   436→              <Money value={p.amount} currency={currency} />
   437→            </span>
   438→            {(p.note || p.auto) && (
   439→              <span className="min-w-0 flex-1 truncate text-[10px] text-mute">
   440→                {p.auto === "tick" ? "Ticked" : p.auto === "settle" ? "Settle up" : p.note}
   441→              </span>
   442→            )}
   443→            <IconButton
   444→              label={`Remove payment ${i + 1}`}
   445→              className="h-9 w-9 shrink-0"
   446→              onClick={() => void remove(p, i)}
   447→            >
   448→              <X size={13} />
   449→            </IconButton>
   450→          </li>
   451→        ))}
   452→      </ol>
   453→      {lt > 0.005 && n > 0 && <ProgressBar className="mt-2" value={(paid / lt) * 100} />}
   454→      <p className="mt-1.5 text-[10.5px] text-mute">
   455→        {n} payment{n !== 1 ? "s" : ""} ·{" "}
   456→        {rem > 0.005 ? (
   457→          <>
   458→            <Money value={rem} currency={currency} /> left
   459→          </>
   460→        ) : rem < -0.005 ? (
   461→          "overpaid"
   462→        ) : (
   463→          "all paid ✓"
   464→        )}
   465→      </p>
   466→      <Button size="sm" variant="soft" className="mt-2 w-full" onClick={() => setPayOpen(true)}>
   467→        <Plus size={13} /> Add payment
   468→      </Button>
   469→      <AddPaymentSheet
   470→        open={payOpen}
   471→        onClose={() => setPayOpen(false)}
   472→        remaining={Math.max(rem, 0)}
   473→        currency={currency}
   474→        onSubmit={(p) => {
   475→          addItemPayment(listId, item.id, p);
   476→          toastOk("Payment added");
   477→          setPayOpen(false);
   478→        }}
   479→      />
   480→    </div>
   481→  );
   482→}

   488→function ItemDiscountField({ listId, item }: { listId: ID; item: ListItem }) {
   489→  const updateListItem = useApp((s) => s.updateListItem);
   490→  const [type, setType] = useState<ListDiscount["type"]>(item.discount?.type ?? "pct");

   494→  };
   495→
   496→  return (
   497→    <div className="flex w-full items-center gap-2">
   498→      <Segmented
   499→        className="w-[116px] shrink-0"
   500→        options={[

   502→          { value: "amount", label: "Amt" },
   503→        ]}
   504→        value={type}
   505→        onChange={(t) => {
   506→          setType(t);
   507→          commit(t, item.discount?.value ?? 0);
   508→        }}
   509→      />
   510→      <NumField
   511→        value={item.discount?.value ?? 0}
   512→        ariaLabel="Row discount value"
   513→        onCommit={(v) => commit(type, v)}
   514→      />
   515→    </div>
   516→  );

   522→
   523→function CalcItemRow({
   524→  listId,
   525→  item,
   526→  expanded,
   527→  onToggleExpand,
   528→  currency,
   529→}: {
   530→  listId: ID;
   531→  item: ListItem;
   532→  expanded: boolean;
   533→  onToggleExpand: () => void;
   534→  currency: string;
   535→}) {
   536→  const confirm = useConfirm();
   537→  const toggleItemChecked = useApp((s) => s.toggleItemChecked);
   538→  const updateListItem = useApp((s) => s.updateListItem);
   539→  const removeListItem = useApp((s) => s.removeListItem);
   540→
   541→  const lt = lineTotal(item);
   542→  const paid = itemPaid(item);
   543→  const rem = itemRemaining(item);
   544→  const tickCount = item.payments.filter((p) => p.auto === "tick").length;
   545→
   546→  const onTick = async (checked: boolean) => {
   547→    if (!checked && tickCount > 0) {
   548→      const ok = await confirm({
   549→        title: "Remove tick payments?",
   550→        message: `Ticking added ${tickCount} payment${tickCount !== 1 ? "s" : ""}. Unticking will remove them.`,
   551→        confirmLabel: "Remove",
   552→        tone: "danger",
   553→      });
   554→      if (!ok) return; // checkbox stays checked — it is store-controlled
   555→    }
   556→    toggleItemChecked(listId, item.id, checked);
   557→  };
   558→
   559→  const removeItem = async () => {
   560→    const ok = await confirm({
   561→      title: `Remove “${item.name}”?`,
   562→      message: "The row and its payment history are removed from this list.",
   563→      confirmLabel: "Remove",
   564→      tone: "danger",
   565→    });
   566→    if (!ok) return;
   567→    removeListItem(listId, item.id);
   568→    toastOk("Item removed");
   569→  };
   570→
   571→  return (
   572→    <div className="rounded-2xl border border-line bg-surface shadow-key">
   573→      {/* collapsed */}
   574→      <div className="flex items-center gap-1 p-2.5 pl-1.5">
   575→        <TickCheckbox checked={item.checked} onChange={(v) => void onTick(v)} name={item.name} />
   576→        <button
   577→          type="button"
   578→          onClick={onToggleExpand}
   579→          aria-expanded={expanded}
   580→          aria-label={`${item.name}. Tap to toggle row details.`}
   581→          className="min-w-0 flex-1 py-1 text-left"
   582→        >
   583→          <span className="flex items-baseline justify-between gap-2">
   584→            <span
   585→              className={cx(
   586→                "truncate text-[12.5px] font-semibold text-ink",
   587→                item.checked && "line-through decoration-line2"
   588→              )}
   589→            >
   590→              {item.name}
   591→            </span>
   592→            <span className="shrink-0 text-[12.5px] font-bold text-ink">
   593→              <Money value={lt} currency={currency} />
   594→            </span>
   595→          </span>
   596→          <span className="mt-0.5 flex items-baseline justify-between gap-2">
   597→            <span className="truncate text-[10.5px] tabular-nums text-mute">
   598→              {item.qty}
   599→              {item.unit ? ` ${item.unit}` : ""} × {item.unitPrice.toLocaleString("en-US")}

   746→function PlainItems({ list }: { list: CalcList }) {
   747→  const confirm = useConfirm();
   748→  const toggleItemChecked = useApp((s) => s.toggleItemChecked);
   749→  const removeListItem = useApp((s) => s.removeListItem);
   750→
   751→  const done = list.items.filter((i) => i.checked).length;
   752→  const total = list.items.length;
   753→
   754→  const onRemove = async (item: ListItem) => {
   755→    const ok = await confirm({
   756→      title: `Remove “${item.name}”?`,
   757→      message: "The row is removed from this list.",
   758→      confirmLabel: "Remove",
   759→      tone: "danger",
   760→    });
   761→    if (!ok) return;
   762→    removeListItem(list.id, item.id);
   763→    toastOk("Item removed");
   764→  };
   765→
   766→  const onTick = async (item: ListItem, checked: boolean) => {
   767→    const tickCount = item.payments.filter((p) => p.auto === "tick").length;
   768→    if (!checked && tickCount > 0) {
   769→      const ok = await confirm({
   770→        title: "Remove tick payments?",
   771→        message: `Ticking added ${tickCount} payment${tickCount !== 1 ? "s" : ""}. Unticking will remove them.`,
   772→        confirmLabel: "Remove",
   773→        tone: "danger",
   774→      });
   775→      if (!ok) return;
   776→    }
   777→    toggleItemChecked(list.id, item.id, checked);
   778→  };
   779→
   780→  return (
   781→    <div className="space-y-2">
   782→      {list.kind === "checklist" && total > 0 && (
   783→        <div className="rounded-[22px] border border-line bg-surface p-4 shadow-key">
   784→          <div className="flex items-center justify-between text-[11px] font-semibold text-mute">
   785→            <span>
   786→              {done} of {total} done
   787→            </span>
   788→            <span className="tabular-nums">{Math.round((done / total) * 100)}%</span>
   789→          </div>
   790→          <ProgressBar className="mt-2" value={(done / total) * 100} />
   791→        </div>
   792→      )}
   793→
   794→      {list.items.map((item) => (
   795→        <div key={item.id} onContextMenu={(e) => { e.preventDefault(); void onRemove(item); }}>
   796→          <SwipeRow onLongPress={() => void onRemove(item)} className="rounded-2xl">
   797→            <div className="flex items-center gap-1 rounded-2xl border border-line bg-surface p-2.5 pl-1.5 shadow-key">
   798→              <TickCheckbox checked={item.checked} onChange={(v) => void onTick(item, v)} name={item.name} />
   799→              <div className="min-w-0 flex-1 py-1">
   800→                <p
   801→                  className={cx(
   802→                    "truncate text-[12.5px] font-semibold text-ink",
   803→                    item.checked && "line-through decoration-line2"
   804→                  )}
   805→                >
   806→                  {item.name}
   807→                </p>
   808→                {item.note && <p className="truncate text-[10.5px] text-mute">{item.note}</p>}
   809→                {list.kind === "todo" && (item.priority || item.dueDate) && (
   810→                  <div className="mt-1 flex flex-wrap items-center gap-1.5">
   811→                    {item.priority && (
   812→                      <span
   813→                        className={cx(
   814→                          "inline-flex shrink-0 items-center rounded-full px-2 py-0.5 text-[10px] font-bold",
   815→                          PRIORITY_META[item.priority].cls
   816→                        )}
   817→                      >
   818→                        {PRIORITY_META[item.priority].label}
   819→                      </span>
   820→                    )}
   821→                    {item.dueDate && <DueBadge dueDate={item.dueDate} />}
   822→                  </div>
   823→                )}
   824→              </div>
   825→            </div>
   826→          </SwipeRow>
   827→        </div>
   828→      ))}
   829→
   830→      {!total && (
   831→        <EmptyState
   832→          icon={<ListPlus size={20} />}
   833→          title="No items yet"
   834→          hint="Add the first item below — long-press any row to remove it."
   835→        />
   836→      )}
   837→    </div>
   838→  );
   839→}
