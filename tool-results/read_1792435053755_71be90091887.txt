    41→interface AppState {

    43→  contacts: Contact[];
    44→  entries: LedgerEntry[];
    45→  lists: CalcList[];
    46→  notes: Note[];
    47→  reminders: Reminder[];
    48→  banks: BankAccount[];
    49→  recents: CallLog[];
    50→  messages: MessageLog[];
    51→  attachments: AttachmentMeta[];
    52→  settings: Settings;
    53→  lastUndo: { label: string; fn: () => void } | null;
    54→
    55→  hydrate: () => Promise<void>;
    56→  pushUndo: (label: string, fn: () => void) => void;
    57→  undo: () => boolean;
    58→
    59→  /* contacts */
    60→  saveContact: (c: Partial<Contact> & { id?: ID }) => Contact;
    61→  deleteContact: (id: ID) => void;
    62→  restoreContact: (id: ID) => void;
    63→  purgeContact: (id: ID) => void;

    65→  mergeContacts: (primaryId: ID, dupId: ID) => void;
    66→  findDuplicates: () => { primary: Contact; dup: Contact; reason: string }[];
    67→
    68→  /* ledger entries */
    69→  saveEntry: (e: Partial<LedgerEntry> & { id?: ID; ownerId: OwnerId; type: "credit" | "debit" }) => LedgerEntry;
    70→  deleteEntry: (id: ID) => void;
    71→  restoreEntry: (id: ID) => void;
    72→  purgeEntry: (id: ID) => void;
    73→  addPayment: (entryId: ID, payment: Partial<Payment> & { amount: number }) => void;
    74→  removePayment: (entryId: ID, paymentId: ID) => void;
    75→  settleUp: (ownerId: OwnerId) => number;
    76→  moveEntity: (kind: "entry" | "list" | "note" | "reminder" | "bank", id: ID, ownerId: OwnerId) => void;
    77→
    78→  /* lists */
    79→  saveList: (l: Partial<CalcList> & { id?: ID; ownerId: OwnerId; title: string }) => CalcList;
    80→  deleteList: (id: ID) => void;
    81→  restoreList: (id: ID) => void;
    82→  purgeList: (id: ID) => void;
    83→  duplicateList: (id: ID, opts?: { asTemplate?: boolean; fresh?: boolean }) => ID | undefined;
    84→  addListItem: (listId: ID, item: Partial<CalcList["items"][number]>) => void;
    85→  updateListItem: (listId: ID, itemId: ID, patch: Partial<CalcList["items"][number]>) => void;
    86→  removeListItem: (listId: ID, itemId: ID) => void;
    87→  addItemPayment: (listId: ID, itemId: ID, payment: Partial<Payment> & { amount: number }) => void;
    88→  removeItemPayment: (listId: ID, itemId: ID, paymentId: ID) => void;
    89→  toggleItemChecked: (listId: ID, itemId: ID, checked: boolean) => void;
    90→  markAllPaid: (listId: ID) => number;
    91→  clearPaid: (listId: ID) => void;
    92→
    93→  /* notes */
    94→  saveNote: (n: Partial<Note> & { id?: ID; ownerId: OwnerId }) => Note;
    95→  deleteNote: (id: ID) => void;
    96→  restoreNote: (id: ID) => void;
    97→  purgeNote: (id: ID) => void;
    98→
    99→  /* reminders */
   100→  saveReminder: (r: Partial<Reminder> & { id?: ID; ownerId: OwnerId; title: string }) => Reminder;
   101→  deleteReminder: (id: ID) => void;
   102→  restoreReminder: (id: ID) => void;
   103→  purgeReminder: (id: ID) => void;
   104→  completeReminder: (id: ID) => void;
   105→  reopenReminder: (id: ID) => void;
   106→  snoozeReminder: (id: ID, until: number) => void;
   107→  fireReminder: (id: ID) => void;
   108→
   109→  /* banks */
   110→  saveBank: (b: Partial<BankAccount> & { id?: ID; ownerId: OwnerId; label: string }) => BankAccount;
   111→  deleteBank: (id: ID) => void;
   112→  restoreBank: (id: ID) => void;
   113→  purgeBank: (id: ID) => void;
   114→
   115→  /* calls + messages */
   116→  addCall: (c: Partial<CallLog> & { phone: string; direction: CallLog["direction"] }) => void;
   117→  deleteRecent: (id: ID) => void;
   118→  clearRecents: () => void;
   119→  logMessage: (ownerId: OwnerId, channel: MessageLog["channel"], text: string) => void;
   120→
   121→  /* attachments */
   122→  addAttachment: (file: File, parentType: AttachmentParentType, parentId: ID) => Promise<AttachmentMeta | null>;
   123→  removeAttachment: (id: ID) => Promise<void>;
   124→  attachmentBlob: (id: ID) => Promise<Blob | undefined>;
   125→
   126→  /* settings + data */
   127→  updateSettings: (patch: Partial<Settings>) => void;
   128→  importContacts: (list: Partial<Contact>[], mode: "skip" | "merge") => { added: number; merged: number; skipped: number };
   129→  wipeAll: () => Promise<void>;
   130→}

   145→
   146→export const useApp = create<AppState>((set, get) => {
   147→  /* ---- internal helpers ---- */
   148→  const touch = <T extends { updatedAt: number }>(e: T): T => ({ ...e, updatedAt: now() });
   149→
   150→  const persistContact = (c: Contact) => void idbPut("contacts", c);
   151→  const persistEntry = (e: LedgerEntry) => void idbPut("entries", e);
   152→  const persistList = (l: CalcList) => void idbPut("lists", l);
   153→  const persistNote = (n: Note) => void idbPut("notes", n);
   154→  const persistReminder = (r: Reminder) => void idbPut("reminders", r);
   155→  const persistBank = (b: BankAccount) => void idbPut("banks", b);
   156→  const persistRecent = (c: CallLog) => void idbPut("recents", c);
   157→  const persistMessage = (m: MessageLog) => void idbPut("messages", m);
   158→  const persistAttachment = (a: AttachmentMeta) => void idbPut("attachments", a);
   159→  const persistSettings = (s: Settings) => void metaSet("settings", s);
   160→
   161→  const updateEntry = (id: ID, fn: (e: LedgerEntry) => LedgerEntry) => {
   162→    const e = get().entries.find((x) => x.id === id);
   163→    if (!e) return;
   164→    const next = touch(fn(e));
   165→    set((s) => ({ entries: s.entries.map((x) => (x.id === id ? next : x)) }));
   166→    persistEntry(next);
   167→  };
   168→
   169→  const updateList = (id: ID, fn: (l: CalcList) => CalcList) => {
   170→    const l = get().lists.find((x) => x.id === id);
   171→    if (!l) return;
   172→    const next = touch(fn(l));
   173→    set((s) => ({ lists: s.lists.map((x) => (x.id === id ? next : x)) }));
   174→    persistList(next);
   175→  };
   176→
   177→  const updateReminder = (id: ID, fn: (r: Reminder) => Reminder) => {
   178→    const r = get().reminders.find((x) => x.id === id);
   179→    if (!r) return;
   180→    const next = touch(fn(r));
   181→    set((s) => ({ reminders: s.reminders.map((x) => (x.id === id ? next : x)) }));
   182→    persistReminder(next);
   183→  };
   184→
   185→  return {
   186→    hydrated: false,
   187→    contacts: [],
   188→    entries: [],
   189→    lists: [],
   190→    notes: [],
   191→    reminders: [],
   192→    banks: [],
   193→    recents: [],
   194→    messages: [],
   195→    attachments: [],
   196→    settings: DEFAULT_SETTINGS,
   197→    lastUndo: null,
   198→
   199→    /* ---------------- hydrate ---------------- */
   200→    hydrate: async () => {
   201→      if (get().hydrated) return;
   202→      const [contacts, entries, lists, notes, reminders, banks, recents, messages, attachments] =
   203→        await Promise.all([
   204→          idbGetAll<Contact>("contacts"),
   205→          idbGetAll<LedgerEntry>("entries"),
   206→          idbGetAll<CalcList>("lists"),
   207→          idbGetAll<Note>("notes"),
   208→          idbGetAll<Reminder>("reminders"),
   209→          idbGetAll<BankAccount>("banks"),
   210→          idbGetAll<CallLog>("recents"),
   211→          idbGetAll<MessageLog>("messages"),
   212→          idbGetAll<AttachmentMeta>("attachments"),
   213→        ]);
   214→      let settings = (await metaGet<Settings>("settings")) || DEFAULT_SETTINGS;
   215→      settings = { ...DEFAULT_SETTINGS, ...settings };
   216→
   217→      const seeded = await metaGet<boolean>("demoSeeded");
   218→      if (!seeded && contacts.length === 0 && entries.length === 0) {
   219→        const demo = buildDemoData();
   220→        set({
   221→          contacts: demo.contacts,
   222→          entries: demo.entries,
   223→          lists: demo.lists,
   224→          notes: demo.notes,
   225→          reminders: demo.reminders,
   226→          banks: demo.banks,
   227→          recents: demo.recents,
   228→          messages: [],
   229→          attachments: [],
   230→          settings: { ...settings, demoSeeded: true },
   231→        });
   232→        await metaSet("demoSeeded", true);
   233→        demo.contacts.forEach(persistContact);
   234→        demo.entries.forEach(persistEntry);
   235→        demo.lists.forEach(persistList);
   236→        demo.notes.forEach(persistNote);
   237→        demo.reminders.forEach(persistReminder);
   238→        demo.banks.forEach(persistBank);
   239→        demo.recents.forEach(persistRecent);
   240→        await metaSet("settings", { ...settings, demoSeeded: true });
   241→        set({ hydrated: true });
   242→        return;
   243→      }
   244→
   245→      set({
   246→        contacts,
   247→        entries,
   248→        lists,
   249→        notes,
   250→        reminders,
   251→        banks,
   252→        recents: recents.sort((a, b) => b.at - a.at),
   253→        messages,
   254→        attachments,
   255→        settings,
   256→        hydrated: true,
   257→      });
   258→    },
   259→
   260→    pushUndo: (label, fn) => set({ lastUndo: { label, fn } }),
   261→    undo: () => {
   262→      const u = get().lastUndo;
   263→      if (!u) return false;
   264→      set({ lastUndo: null });
   265→      u.fn();
   266→      return true;
   267→    },
   268→
   269→    /* ---------------- contacts ---------------- */
   270→    saveContact: (input) => {
   271→      const existing = input.id ? get().contacts.find((c) => c.id === input.id) : undefined;
   272→      const id = input.id || uid();
   273→      const c: Contact = {
   274→        id,
   275→        firstName: input.firstName ?? existing?.firstName ?? "",
   276→        lastName: input.lastName ?? existing?.lastName ?? "",
   277→        nickname: input.nickname ?? existing?.nickname,
   278→        company: input.company ?? existing?.company,
   279→        jobTitle: input.jobTitle ?? existing?.jobTitle,
   280→        phones: input.phones ?? existing?.phones ?? [],
   281→        emails: input.emails ?? existing?.emails ?? [],
   282→        links: input.links ?? existing?.links ?? [],
   283→        socials: input.socials ?? existing?.socials ?? [],
   284→        addresses: input.addresses ?? existing?.addresses ?? [],
   285→        customFields: input.customFields ?? existing?.customFields ?? [],
   286→        birthday: input.birthday ?? existing?.birthday,
   287→        tags: input.tags ?? existing?.tags ?? [],
   288→        favorite: input.favorite ?? existing?.favorite ?? false,
   289→        pinnedNote: input.pinnedNote ?? existing?.pinnedNote,
   290→        note: input.note ?? existing?.note,
   291→        photoId: input.photoId ?? existing?.photoId,
   292→        color: input.color ?? existing?.color,
   293→        textColor: input.textColor ?? existing?.textColor,
   294→        speedDial: input.speedDial ?? existing?.speedDial,
   295→        createdAt: existing?.createdAt ?? now(),
   296→        updatedAt: now(),
   297→      };
   298→      // only one primary phone/email
   299→      if (c.phones.some((p) => p.primary)) c.phones.forEach((p, i) => { if (i > 0 && p.primary && c.phones[0].primary) p.primary = false; });
   300→      set((s) => ({
   301→        contacts: existing
   302→          ? s.contacts.map((x) => (x.id === id ? c : x))
   303→          : [...s.contacts, c],
   304→      }));
   305→      persistContact(c);
   306→      return c;
   307→    },
   308→
   309→    deleteContact: (id) => {
   310→      const c = get().contacts.find((x) => x.id === id);
   311→      if (!c) return;
   312→      const next = { ...c, deletedAt: now() };
   313→      set((s) => ({ contacts: s.contacts.map((x) => (x.id === id ? next : x)) }));
   314→      persistContact(next);
   315→      get().pushUndo(`“${displayName(c)}” moved to Trash`, () => get().restoreContact(id));
   316→    },
   317→
   318→    restoreContact: (id) => {
   319→      const c = get().contacts.find((x) => x.id === id);
   320→      if (!c) return;
   321→      const next = { ...c, deletedAt: undefined };
   322→      set((s) => ({ contacts: s.contacts.map((x) => (x.id === id ? next : x)) }));
   323→      persistContact(next);
   324→    },
   325→
   326→    purgeContact: async (id) => {
   327→      set((s) => ({ contacts: s.contacts.filter((x) => x.id !== id) }));
   328→      await idbDel("contacts", id);
   329→    },
   330→
   331→    toggleFavorite: (id) => {
   332→      const c = get().contacts.find((x) => x.id === id);
   333→      if (!c) return;
   334→      const next = touch({ ...c, favorite: !c.favorite });
   335→      set((s) => ({ contacts: s.contacts.map((x) => (x.id === id ? next : x)) }));
   336→      persistContact(next);
   337→    },
   338→
   339→    mergeContacts: (primaryId, dupId) => {
   340→      const { contacts, entries, lists, notes, reminders, banks, recents, attachments } = get();
   341→      const primary = contacts.find((c) => c.id === primaryId);
   342→      const dup = contacts.find((c) => c.id === dupId);
   343→      if (!primary || !dup) return;
   344→
   345→      const unionById = <T extends { id: string }>(a: T[], b: T[]) => {
   346→        const ids = new Set(a.map((x) => x.id));
   347→        return [...a, ...b.filter((x) => !ids.has(x.id))];
   348→      };
   349→      const phoneKeys = new Set(primary.phones.map((p) => normalizePhone(p.number)));
   350→      const merged: Contact = touch({
   351→        ...primary,
   352→        phones: [
   353→          ...primary.phones,
   354→          ...dup.phones.filter((p) => !phoneKeys.has(normalizePhone(p.number))),
   355→        ],
   356→        emails: unionById(primary.emails, dup.emails),
   357→        links: unionById(primary.links, dup.links),
   358→        socials: unionById(primary.socials, dup.socials),
   359→        addresses: unionById(primary.addresses, dup.addresses),
   360→        customFields: unionById(primary.customFields, dup.customFields),
   361→        tags: [...new Set([...primary.tags, ...dup.tags])],
   362→        favorite: primary.favorite || dup.favorite,
   363→        birthday: primary.birthday || dup.birthday,
   364→        note: [primary.note, dup.note].filter(Boolean).join("\n") || undefined,
   365→      });
   366→
   367→      const remapOwner = <T extends { ownerId: OwnerId }>(arr: T[], fn: (x: T) => void) =>
   368→        arr.map((x) => {
   369→          if (x.ownerId === dupId) {
   370→            const n = { ...x, ownerId: primaryId, updatedAt: now() };
   371→            fn(n as T);
   372→            return n;
   373→          }
   374→          return x;
   375→        });
   376→
   377→      const nextEntries = remapOwner(entries, (e) => persistEntry(e));
   378→      const nextLists = remapOwner(lists, (l) => persistList(l));
   379→      const nextNotes = remapOwner(notes, (n) => persistNote(n));
   380→      const nextReminders = remapOwner(reminders, (r) => persistReminder(r));
   381→      const nextBanks = remapOwner(banks, (b) => persistBank(b));
   382→      const nextRecents = recents.map((r) =>
   383→        r.contactId === dupId ? (persistRecent({ ...r, contactId: primaryId }), { ...r, contactId: primaryId }) : r
   384→      );
   385→      const nextAttachments = attachments.map((a) =>
   386→        a.parentType === "contact" && a.parentId === dupId
   387→          ? (persistAttachment({ ...a, parentId: primaryId }), { ...a, parentId: primaryId })
   388→          : a
   389→      );
   390→
   391→      set({
   392→        contacts: contacts.map((x) => (x.id === primaryId ? merged : x)).filter((x) => x.id !== dupId),
   393→        entries: nextEntries,
   394→        lists: nextLists,
   395→        notes: nextNotes,
   396→        reminders: nextReminders,
   397→        banks: nextBanks,
   398→        recents: nextRecents,
   399→        attachments: nextAttachments,
   400→      });
   401→      persistContact(merged);
   402→      void idbDel("contacts", dupId);
   403→    },
   404→
   405→    findDuplicates: () => {
   406→      const { contacts } = get();
   407→      const live = contacts.filter((c) => !c.deletedAt);
   408→      const results: { primary: Contact; dup: Contact; reason: string }[] = [];
   409→      const seen = new Set<ID>();
   410→      for (let i = 0; i < live.length; i++) {
   411→        for (let j = i + 1; j < live.length; j++) {
   412→          const a = live[i];
   413→          const b = live[j];
   414→          if (seen.has(a.id) || seen.has(b.id)) continue;
   415→          const aPhones = new Set(a.phones.map((p) => normalizePhone(p.number)));
   416→          const sharedPhone = b.phones.some((p) => aPhones.has(normalizePhone(p.number)));
   417→          const sameName =
   418→            displayName(a).toLowerCase() === displayName(b).toLowerCase() && displayName(a) !== "Unknown";
   419→          const aEmails = new Set(a.emails.map((e) => e.address.toLowerCase()));
   420→          const sharedEmail = b.emails.some((e) => aEmails.has(e.address.toLowerCase()));
   421→          if (sharedPhone || (sameName && sharedEmail)) {
   422→            results.push({
   423→              primary: a,
   424→              dup: b,
   425→              reason: sharedPhone ? "Same phone number" : "Same name and email",
   426→            });
   427→            seen.add(b.id);
   428→          }
   429→        }
   430→      }
   431→      return results;
   432→    },
   433→
   434→    /* ---------------- ledger entries ---------------- */
   435→    saveEntry: (input) => {
   436→      const existing = input.id ? get().entries.find((e) => e.id === input.id) : undefined;
   437→      const id = input.id || uid();
   438→      const e: LedgerEntry = {
   439→        id,
   440→        ownerId: input.ownerId,
   441→        type: input.type,
   442→        amount: r2Fallback(input.amount ?? existing?.amount ?? 0),
   443→        payments: input.payments ?? existing?.payments ?? [],
   444→        date: input.date ?? existing?.date ?? todayStr(),
   445→        dueDate: input.dueDate ?? existing?.dueDate,
   446→        category: input.category ?? existing?.category,
   447→        method: input.method ?? existing?.method,
   448→        note: input.note ?? existing?.note,
   449→        listId: input.listId ?? existing?.listId,
   450→        attachments: input.attachments ?? existing?.attachments ?? [],
   451→        createdAt: existing?.createdAt ?? now(),
   452→        updatedAt: now(),
   453→      };
   454→      set((s) => ({
   455→        entries: existing ? s.entries.map((x) => (x.id === id ? e : x)) : [...s.entries, e],
   456→      }));
   457→      persistEntry(e);
   458→      return e;
   459→    },
   460→
   461→    deleteEntry: (id) => {
   462→      const e = get().entries.find((x) => x.id === id);
   463→      if (!e) return;
   464→      const next = { ...e, deletedAt: now() };
   465→      set((s) => ({ entries: s.entries.map((x) => (x.id === id ? next : x)) }));
   466→      persistEntry(next);
   467→      get().pushUndo("Entry moved to Trash", () => get().restoreEntry(id));
   468→    },
   469→
   470→    restoreEntry: (id) => {
   471→      const e = get().entries.find((x) => x.id === id);
   472→      if (!e) return;
   473→      const next = { ...e, deletedAt: undefined };
   474→      set((s) => ({ entries: s.entries.map((x) => (x.id === id ? next : x)) }));
   475→      persistEntry(next);
   476→    },
   477→
   478→    purgeEntry: async (id) => {
   479→      set((s) => ({ entries: s.entries.filter((x) => x.id !== id) }));
   480→      await idbDel("entries", id);
   481→    },
   482→
   483→    addPayment: (entryId, payment) => {
   484→      const p: Payment = {
   485→        id: uid(),
   486→        amount: r2Fallback(payment.amount),
   487→        date: payment.date || todayStr(),
   488→        note: payment.note,
   489→        method: payment.method,
   490→        auto: payment.auto,
   491→        createdAt: now(),
   492→      };
   493→      updateEntry(entryId, (e) => ({ ...e, payments: [...e.payments, p] }));
   494→    },
   495→
   496→    removePayment: (entryId, paymentId) => {
   497→      updateEntry(entryId, (e) => ({
   498→        ...e,
   499→        payments: e.payments.filter((p) => p.id !== paymentId),
   500→      }));
   501→    },
   502→
   503→    settleUp: (ownerId) => {
   504→      const balance = calc.ownerBalance(get().entries, ownerId);
   505→      if (Math.abs(balance) < 0.005) return 0;
   506→      let left = Math.abs(balance);
   507→      const ordered = get()
   508→        .entries.filter((e) => e.ownerId === ownerId && !e.deletedAt)
   509→        .sort((a, b) => a.date.localeCompare(b.date) || a.createdAt - b.createdAt);
   510→      for (const e of ordered) {
   511→        if (left < 0.005) break;
   512→        const rem = calc.entryRemaining(e);
   513→        // paying down: reduce |remaining| in the direction that zeroes the balance
   514→        const effective = e.type === "credit" ? rem : -rem;

  1089→async function compressImage(
  1090→  file: File,
  1091→  maxSide: number,
  1092→  quality: number
  1093→): Promise<{ blob: Blob; width: number; height: number }> {
  1094→  const img = await new Promise<HTMLImageElement>((resolve, reject) => {
  1095→    const url = URL.createObjectURL(file);
  1096→    const i = new Image();
  1097→    i.onload = () => {
  1098→      URL.revokeObjectURL(url);
  1099→      resolve(i);
  1100→    };
  1101→    i.onerror = () => {
  1102→      URL.revokeObjectURL(url);
  1103→      reject(new Error("bad image"));
  1104→    };
  1105→    i.src = url;
  1106→  });
  1107→  const scale = Math.min(1, maxSide / Math.max(img.width, img.height));
  1108→  const w = Math.max(1, Math.round(img.width * scale));
  1109→  const h = Math.max(1, Math.round(img.height * scale));
  1110→  const canvas = document.createElement("canvas");
  1111→  canvas.width = w;
  1112→  canvas.height = h;
  1113→  const ctx = canvas.getContext("2d");
  1114→  if (!ctx) return { blob: file, width: img.width, height: img.height };
  1115→  ctx.drawImage(img, 0, 0, w, h);
  1116→  const blob = await new Promise<Blob>((resolve) =>
  1117→    canvas.toBlob((b) => resolve(b || file), "image/jpeg", quality)
  1118→  );
  1119→  return { blob, width: w, height: h };
  1120→}
