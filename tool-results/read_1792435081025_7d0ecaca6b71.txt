   132→async function shareVCard(c: Contact) {
   133→  const vcf = toVCard([c], true);
   134→  const res = await shareText(`Contact · ${displayName(c)}`, vcf);
   135→  if (res === "copied") {
   136→    await copyText(vcf);
   137→    toastOk("vCard copied to clipboard");
   138→  } else if (res === "failed") {
   139→    downloadBlob(new Blob([vcf], { type: "text/vcard" }), `${displayName(c)}.vcf`);
   140→    toastOk("vCard downloaded");
   141→  }
   142→}

   144→function daysUntilBirthday(bd?: string): number | null {
   145→  if (!bd) return null;
   146→  const d = new Date(`${bd}T00:00:00`);
   147→  if (isNaN(d.getTime())) return null;
   148→  const today = new Date();
   149→  today.setHours(0, 0, 0, 0);
   150→  let next = new Date(today.getFullYear(), d.getMonth(), d.getDate());
   151→  if (next.getTime() < today.getTime()) next = new Date(today.getFullYear() + 1, d.getMonth(), d.getDate());
   152→  return Math.round((next.getTime() - today.getTime()) / 86400000);
   153→}

   245→function socialUrl(s: Social): string | null {
   246→  const h = s.handle.trim().replace(/^@/, "");
   247→  switch (s.app) {
   248→    case "telegram":
   249→      return tgLink(s.handle);
   250→    case "whatsapp":
   251→      return waLink(h);
   252→    case "instagram":
   253→      return `https://instagram.com/${encodeURIComponent(h)}`;
   254→    case "x":
   255→      return `https://x.com/${encodeURIComponent(h)}`;
   256→    case "signal":
   257→      return `https://signal.me/#p/${encodeURIComponent(h)}`;
   258→    default:
   259→      return /^https?:\/\//i.test(h) ? h : null;
   260→  }
   261→}

  1595→export function CallScreen({
  1596→  phone,
  1597→  contactId,
  1598→  direction,
  1599→}: {
  1600→  phone: string;
  1601→  contactId?: ID;
  1602→  direction?: "incoming" | "outgoing" | "missed";
  1603→}) {
  1604→  const app = useApp();
  1605→  const ui = useUI();
  1606→  const cc = app.settings.countryCode;
  1607→
  1608→  /* hide while another overlay sits on top of the call (e.g. adding a contact mid-call) */
  1609→  const stack = useUI((s) => s.overlayStack);
  1610→  const isTop = stack.length > 0 && stack[stack.length - 1].type === "call";
  1611→
  1612→  const contact = useMemo(() => {
  1613→    const byId = contactId ? app.contacts.find((c) => c.id === contactId && !c.deletedAt) : undefined;
  1614→    if (byId) return byId;
  1615→    const norm = normalizePhone(phone, cc);
  1616→    return (
  1617→      app.contacts.find((c) =>
  1618→        !c.deletedAt && c.phones.some((p) => normalizePhone(p.number, cc) === norm)
  1619→      ) ?? undefined
  1620→    );
  1621→  }, [app.contacts, contactId, phone, cc]);
  1622→
  1623→  const [phase, setPhase] = useState<CallPhase>(direction === "incoming" ? "incoming" : "dialing");
  1624→  const [seconds, setSeconds] = useState(0);
  1625→  const [muted, setMuted] = useState(false);
  1626→  const [speaker, setSpeaker] = useState(false);
  1627→  const [keypadOn, setKeypadOn] = useState(false);
  1628→  const [dtmf, setDtmf] = useState("");
  1629→  const [closePending, setClosePending] = useState(false);
  1630→  const finalizedRef = useRef(false);
  1631→  const isTopRef = useRef(isTop);
  1632→  useEffect(() => {
  1633→    isTopRef.current = isTop;
  1634→  }, [isTop]);
  1635→
  1636→  /* dialing → connected after 1.5s */
  1637→  useEffect(() => {
  1638→    if (phase !== "dialing") return;
  1639→    const t = setTimeout(() => setPhase("connected"), 1500);
  1640→    return () => clearTimeout(t);
  1641→  }, [phase]);
  1642→
  1643→  /* live timer */
  1644→  useEffect(() => {
  1645→    if (phase !== "connected") return;
  1646→    const i = setInterval(() => setSeconds((s) => s + 1), 1000);
  1647→    return () => clearInterval(i);
  1648→  }, [phase]);
  1649→
  1650→  const logAndClose = (dur: number, dir: CallLog["direction"]) => {
  1651→    if (finalizedRef.current) return;
  1652→    finalizedRef.current = true;
  1653→    app.addCall({ phone, contactId: contact?.id ?? contactId, direction: dir, duration: dur });
  1654→    if (isTopRef.current) ui.pop();
  1655→    else setClosePending(true);
  1656→  };
  1657→
  1658→  /* "Call ended" shows briefly, then the screen closes */
  1659→  useEffect(() => {
